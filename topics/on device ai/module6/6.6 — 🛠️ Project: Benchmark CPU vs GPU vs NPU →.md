## 🎯 Project Goal

Build a **production-quality hardware benchmark Android app** that:

- Runs YOLOv8n TFLite across **CPU, GPU, and NNAPI/NPU** backends
- Shows real-time inference on the live camera while benchmarking
- Displays animated bar charts comparing latency and FPS per backend
- Reports device specs (chip, RAM, NPU support) alongside results
- Exports a shareable benchmark report as a text/markdown file
- Handles unsupported backends gracefully with clear user feedback

## 3. DeviceInfo.kt — Collect Device Specs

```kotlin
// data/DeviceInfo.kt
package com.yourname.benchmark.data

import android.app.ActivityManager
import android.content.Context
import android.os.Build
import android.util.Log
import org.tensorflow.lite.gpu.CompatibilityList
import org.tensorflow.lite.nnapi.NnApiDelegate

data class DeviceInfo(
    val manufacturer: String,
    val model: String,
    val chipset: String,
    val androidVersion: Int,
    val totalRamGb: Float,
    val availableRamGb: Float,
    val gpuCompatible: Boolean,
    val nnApiLevel: Int,
    val nnApiDevices: List<String>,
    val abis: List<String>
) {
    val displayName get() = "$manufacturer $model"

    val npuStatus get() = when {
        nnApiDevices.any { it.contains("hta", true) || it.contains("htp", true) }
            -> "✅ Hexagon NPU"
        nnApiDevices.any { it.contains("neurapilot", true) || it.contains("npu", true) }
            -> "✅ MediaTek NPU"
        nnApiDevices.any { it.contains("tpu", true) || it.contains("tensor", true) }
            -> "✅ Google TPU"
        nnApiDevices.isNotEmpty() && !nnApiDevices.all { it.contains("cpu", true) }
            -> "⚠️ NPU (vendor)"
        else -> "❌ No NPU"
    }

    fun toMarkdown(): String = buildString {
        appendLine("### Device Specs")
        appendLine("| Property | Value |")
        appendLine("|---|---|")
        appendLine("| Device | $displayName |")
        appendLine("| Android | API $androidVersion |")
        appendLine("| Total RAM | ${"%.1f".format(totalRamGb)} GB |")
        appendLine("| Available RAM | ${"%.1f".format(availableRamGb)} GB |")
        appendLine("| GPU | ${if (gpuCompatible) "✅ Compatible" else "❌ Not compatible"} |")
        appendLine("| NPU | $npuStatus |")
        appendLine("| NNAPI Level | $nnApiLevel |")
        appendLine("| Accelerators | ${nnApiDevices.joinToString(", ")} |")
    }

    companion object {
        fun collect(context: Context): DeviceInfo {
            val actManager = context.getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
            val memInfo = ActivityManager.MemoryInfo().also { actManager.getMemoryInfo(it) }

            val nnApiDevices = mutableListOf<String>()
            try {
                val delegate = NnApiDelegate()
                // Collect available device names via reflection or API
                nnApiDevices.addAll(listOf("nnapi-reference"))  // placeholder
                delegate.close()
            } catch (e: Exception) {
                Log.w("DeviceInfo", "NNAPI query failed: ${e.message}")
            }

            return DeviceInfo(
                manufacturer   = Build.MANUFACTURER,
                model          = Build.MODEL,
                chipset        = Build.HARDWARE,
                androidVersion = Build.VERSION.SDK_INT,
                totalRamGb     = memInfo.totalMem / 1e9f,
                availableRamGb = memInfo.availMem / 1e9f,
                gpuCompatible  = CompatibilityList().isDelegateSupportedOnThisDevice,
                nnApiLevel     = Build.VERSION.SDK_INT,
                nnApiDevices   = nnApiDevices,
                abis           = Build.SUPPORTED_ABIS.toList()
            )
        }
    }
}
```

```kotlin
// data/BenchmarkResult.kt
package com.yourname.benchmark.data

enum class Backend(val displayName: String, val emoji: String) {
    CPU  ("CPU (4 threads)", "🔲"),
    GPU  ("GPU (FP16)",      "🎮"),
    NNAPI("NNAPI (NPU)",     "⚡")
}

data class BackendBenchmarkResult(
    val backend: Backend,
    val avgMs: Float,
    val minMs: Long,
    val maxMs: Long,
    val p95Ms: Long,
    val fps: Float,
    val supported: Boolean,
    val errorMessage: String = "",
    val detections: Int = 0          // avg detections per frame
)

data class BenchmarkSession(
    val deviceInfo: DeviceInfo,
    val modelName: String,
    val results: List<BackendBenchmarkResult>,
    val timestampMs: Long = System.currentTimeMillis()
) {
    val fastestResult get() =
        results.filter { it.supported }.minByOrNull { it.avgMs }

    val cpuResult  get() = results.find { it.backend == Backend.CPU }
    val gpuResult  get() = results.find { it.backend == Backend.GPU }
    val nnapiResult get() = results.find { it.backend == Backend.NNAPI }

    fun speedupVsCpu(result: BackendBenchmarkResult): Float {
        val cpuAvg = cpuResult?.avgMs ?: return 1f
        return if (result.avgMs > 0) cpuAvg / result.avgMs else 0f
    }

    fun toMarkdown(): String = buildString {
        appendLine("# TFLite Benchmark Report")
        appendLine()
        appendLine(deviceInfo.toMarkdown())
        appendLine()
        appendLine("### Results: $modelName")
        appendLine()
        appendLine("| Backend | Avg | Min | Max | P95 | FPS | vs CPU |")
        appendLine("|---|---|---|---|---|---|---|")

        results.filter { it.supported }.forEach { r ->
            val speedup = speedupVsCpu(r)
            val speedupStr = if (r.backend == Backend.CPU) "1×"
                             else "${"%.1f".format(speedup)}× faster"
            appendLine(
                "| ${r.backend.emoji} ${r.backend.displayName} " +
                "| ${r.avgMs.toInt()}ms " +
                "| ${r.minMs}ms " +
                "| ${r.maxMs}ms " +
                "| ${r.p95Ms}ms " +
                "| ${r.fps.toInt()} fps " +
                "| $speedupStr |"
            )
        }

        results.filter { !it.supported }.forEach { r ->
            appendLine("| ${r.backend.emoji} ${r.backend.displayName} | ❌ ${r.errorMessage} | | | | | |")
        }

        appendLine()
        fastestResult?.let {
            appendLine("🏆 **Fastest backend: ${it.backend.displayName}** " +
                "(${it.avgMs.toInt()}ms · ${it.fps.toInt()} fps)")
        }
    }
}
```

## 5. BackendRunner.kt — Run Model on Each Backend

```kotlin
// ml/BackendRunner.kt
package com.yourname.benchmark.ml

import android.content.Context
import android.graphics.Bitmap
import android.os.SystemClock
import android.util.Log
import com.yourname.benchmark.data.Backend
import com.yourname.benchmark.data.BackendBenchmarkResult
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.tensorflow.lite.Interpreter
import org.tensorflow.lite.gpu.CompatibilityList
import org.tensorflow.lite.gpu.GpuDelegate
import org.tensorflow.lite.nnapi.NnApiDelegate
import org.tensorflow.lite.support.common.FileUtil
import org.tensorflow.lite.support.image.ImageProcessor
import org.tensorflow.lite.support.image.TensorImage
import org.tensorflow.lite.support.image.ops.ResizeOp
import org.tensorflow.lite.support.tensorbuffer.TensorBuffer
import java.nio.ByteBuffer
import java.nio.ByteOrder

class BackendRunner(
    private val context: Context,
    private val modelFileName: String = "yolov8n.tflite",
    private val warmupRuns: Int = 5,
    private val timedRuns: Int = 30,
    private val onProgress: (String) -> Unit = {}
) {
    companion object {
        private const val TAG       = "BackendRunner"
        private const val INPUT_SIZE = 320
    }

    private val modelBuffer by lazy {
        FileUtil.loadMappedFile(context, modelFileName)
    }

    private val imageProcessor = ImageProcessor.Builder()
        .add(ResizeOp(INPUT_SIZE, INPUT_SIZE, ResizeOp.ResizeMethod.BILINEAR))
        .build()

    /**
     * Run benchmark on all three backends.
     * Returns results in order: CPU, GPU, NNAPI.
     */
    suspend fun runAll(testBitmap: Bitmap): List<BackendBenchmarkResult> =
        withContext(Dispatchers.IO) {
            listOf(
                runBackend(Backend.CPU,   testBitmap),
                runBackend(Backend.GPU,   testBitmap),
                runBackend(Backend.NNAPI, testBitmap)
            )
        }

    /**
     * Run benchmark for a specific backend.
     */
    suspend fun runBackend(backend: Backend, testBitmap: Bitmap): BackendBenchmarkResult =
        withContext(Dispatchers.IO) {

            onProgress("Benchmarking ${backend.displayName}...")
            Log.d(TAG, "Starting $backend benchmark ($warmupRuns warmup, $timedRuns timed)")

            var delegate: AutoCloseable? = null

            return@withContext try {
                val options = when (backend) {
                    Backend.CPU -> Interpreter.Options().apply {
                        setNumThreads(4)
                        setUseXNNPACK(true)
                    }

                    Backend.GPU -> {
                        if (!CompatibilityList().isDelegateSupportedOnThisDevice)
                            throw UnsupportedOperationException("GPU not compatible")
                        val gpuDelegate = GpuDelegate(
                            GpuDelegate.Options()
                                .setPrecisionLossAllowed(true)
                                .setInferencePreference(
                                    GpuDelegate.Options.INFERENCE_PREFERENCE_FAST_SINGLE_ANSWER
                                )
                        )
                        delegate = gpuDelegate
                        Interpreter.Options().addDelegate(gpuDelegate)
                    }

                    Backend.NNAPI -> {
                        val nnApiDelegate = NnApiDelegate(
                            NnApiDelegate.Options()
                                .setAllowFp16(true)
                                .setCacheDir(context.cacheDir.absolutePath)
                                .setModelToken("yolov8n_nnapi_bench")
                        )
                        delegate = nnApiDelegate
                        Interpreter.Options().addDelegate(nnApiDelegate)
                    }
                }

                val interpreter = Interpreter(modelBuffer, options)
                val outputShape = interpreter.getOutputTensor(0).shape()
                val outputType  = interpreter.getOutputTensor(0).dataType()
                val outputBuf   = TensorBuffer.createFixedSize(outputShape, outputType)

                // Preprocess test image once (same input every run)
                val tensorImage = imageProcessor.process(TensorImage.fromBitmap(testBitmap))

                // Warmup runs
                repeat(warmupRuns) {
                    interpreter.run(tensorImage.buffer.rewind(), outputBuf.buffer.rewind())
                }

                // Timed runs
                val latencies = mutableListOf<Long>()
                repeat(timedRuns) {
                    val start = SystemClock.elapsedRealtimeNanos()
                    interpreter.run(tensorImage.buffer.rewind(), outputBuf.buffer.rewind())
                    latencies.add((SystemClock.elapsedRealtimeNanos() - start) / 1_000_000)
                }

                interpreter.close()
                latencies.sort()

                val avg = latencies.average().toFloat()
                val fps = if (avg > 0) 1000f / avg else 0f

                Log.d(TAG, "$backend: avg=${avg.toInt()}ms fps=${fps.toInt()}")

                BackendBenchmarkResult(
                    backend   = backend,
                    avgMs     = avg,
                    minMs     = latencies.first(),
                    maxMs     = latencies.last(),
                    p95Ms     = latencies[(latencies.size * 0.95).toInt()],
                    fps       = fps,
                    supported = true
                )

            } catch (e: Exception) {
                Log.w(TAG, "${backend} not supported: ${e.message}")
                BackendBenchmarkResult(
                    backend      = backend,
                    avgMs        = 0f,
                    minMs        = 0,
                    maxMs        = 0,
                    p95Ms        = 0,
                    fps          = 0f,
                    supported    = false,
                    errorMessage = e.message ?: "Unsupported"
                )
            } finally {
                delegate?.close()
            }
        }
}
```

## 6. BenchmarkBarChart.kt — Custom Bar Chart View

```kotlin
// ui/BenchmarkBarChart.kt
package com.yourname.benchmark.ui

import android.animation.ValueAnimator
import android.content.Context
import android.graphics.*
import android.util.AttributeSet
import android.view.View
import android.view.animation.DecelerateInterpolator
import com.yourname.benchmark.data.BackendBenchmarkResult
import com.yourname.benchmark.data.Backend

/**
 * Custom bar chart that animates benchmark results.
 * Bars grow from zero to their final values on first render.
 */
class BenchmarkBarChart @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null
) : View(context, attrs) {

    private var results: List<BackendBenchmarkResult> = emptyList()
    private var animationProgress = 0f

    private val barColors = mapOf(
        Backend.CPU   to Color.parseColor("#4A90D9"),
        Backend.GPU   to Color.parseColor("#27AE60"),
        Backend.NNAPI to Color.parseColor("#E67E22")
    )

    private val barPaint   = Paint(Paint.ANTI_ALIAS_FLAG).apply { style = Paint.Style.FILL }
    private val labelPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color    = Color.parseColor("#333333")
        textSize = 36f
        typeface = Typeface.DEFAULT_BOLD
    }
    private val valuePaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color    = Color.WHITE
        textSize = 32f
        typeface = Typeface.DEFAULT_BOLD
    }
    private val subLabelPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color    = Color.parseColor("#888888")
        textSize = 28f
    }
    private val fastestPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color    = Color.parseColor("#F39C12")
        textSize = 28f
        typeface = Typeface.DEFAULT_BOLD
    }

    fun setResults(results: List<BackendBenchmarkResult>) {
        this.results = results
        animateIn()
    }

    private fun animateIn() {
        ValueAnimator.ofFloat(0f, 1f).apply {
            duration     = 800L
            interpolator = DecelerateInterpolator()
            addUpdateListener { anim ->
                animationProgress = anim.animatedValue as Float
                invalidate()
            }
            start()
        }
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        if (results.isEmpty()) return

        val supported  = results.filter { it.supported }
        if (supported.isEmpty()) return

        val maxAvg     = supported.maxOf { it.avgMs }
        val fastest    = supported.minByOrNull { it.avgMs }

        val paddingLeft  = 220f   // space for backend labels
        val paddingRight = 100f
        val barAreaWidth = width - paddingLeft - paddingRight
        val barHeight    = 72f
        val barGap       = 24f
        val startY       = 40f

        supported.forEachIndexed { i, result ->
            val top    = startY + i * (barHeight + barGap + 40f)
            val color  = barColors[result.backend] ?: Color.GRAY
            val isFastest = result == fastest

            // Backend label
            labelPaint.color = if (isFastest) Color.parseColor("#2C3E50")
                               else Color.parseColor("#555555")
            canvas.drawText(
                result.backend.emoji + " " + result.backend.displayName.take(10),
                10f, top + barHeight * 0.7f, labelPaint
            )

            // Bar background (empty)
            barPaint.color = Color.parseColor("#F0F0F0")
            canvas.drawRoundRect(
                RectF(paddingLeft, top, paddingLeft + barAreaWidth, top + barHeight),
                8f, 8f, barPaint
            )

            // Bar fill (animated)
            val barWidth = (result.avgMs / maxAvg) * barAreaWidth * animationProgress
            barPaint.color = color
            if (barWidth > 0) {
                canvas.drawRoundRect(
                    RectF(paddingLeft, top, paddingLeft + barWidth, top + barHeight),
                    8f, 8f, barPaint
                )
            }

            // Value inside bar
            val valueText = "${result.avgMs.toInt()}ms"
            if (barWidth > valuePaint.measureText(valueText) + 16f) {
                canvas.drawText(valueText, paddingLeft + 12f, top + barHeight * 0.68f, valuePaint)
            }

            // FPS label below bar
            val fpsText = "${result.fps.toInt()} fps"
            canvas.drawText(fpsText, paddingLeft, top + barHeight + 26f, subLabelPaint)

            // Fastest label
            if (isFastest) {
                canvas.drawText(
                    "🏆 FASTEST",
                    paddingLeft + barWidth + 8f,
                    top + barHeight * 0.68f,
                    fastestPaint
                )
            }
        }
    }

    override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
        val n       = results.filter { it.supported }.size.coerceAtLeast(1)
        val desired = (n * (72 + 24 + 40) + 80).toInt()
        super.onMeasure(
            widthMeasureSpec,
            MeasureSpec.makeMeasureSpec(desired, MeasureSpec.EXACTLY)
        )
    }
}
```

```kotlin
// viewmodel/BenchmarkViewModel.kt
package com.yourname.benchmark.viewmodel

import android.app.Application
import android.graphics.Bitmap
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.yourname.benchmark.data.*
import com.yourname.benchmark.ml.BackendRunner
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

data class BenchmarkUiState(
    val deviceInfo: DeviceInfo? = null,
    val session: BenchmarkSession? = null,
    val isRunning: Boolean = false,
    val currentStep: String = "",
    val liveFrame: Bitmap? = null,
    val error: String? = null
)

class BenchmarkViewModel(application: Application) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(BenchmarkUiState())
    val uiState: StateFlow<BenchmarkUiState> = _uiState.asStateFlow()

    private val runner = BackendRunner(
        context      = application,
        warmupRuns   = 5,
        timedRuns    = 30,
        onProgress   = { step ->
            _uiState.update { it.copy(currentStep = step) }
        }
    )

    init {
        collectDeviceInfo()
    }

    private fun collectDeviceInfo() {
        viewModelScope.launch {
            val info = DeviceInfo.collect(getApplication())
            _uiState.update { it.copy(deviceInfo = info) }
        }
    }

    fun runAllBenchmarks(testBitmap: Bitmap) {
        if (_uiState.value.isRunning) return

        _uiState.update { it.copy(isRunning = true, error = null) }

        viewModelScope.launch {
            try {
                val results = runner.runAll(testBitmap)
                val session = BenchmarkSession(
                    deviceInfo = _uiState.value.deviceInfo
                        ?: DeviceInfo.collect(getApplication()),
                    modelName = "YOLOv8n (320×320)",
                    results   = results
                )
                _uiState.update { it.copy(
                    session   = session,
                    isRunning = false,
                    currentStep = "Complete!"
                )}
            } catch (e: Exception) {
                _uiState.update { it.copy(
                    isRunning = false,
                    error     = e.message
                )}
            }
        }
    }

    fun runSingleBackend(backend: Backend, testBitmap: Bitmap) {
        if (_uiState.value.isRunning) return

        _uiState.update { it.copy(isRunning = true) }

        viewModelScope.launch {
            val result  = runner.runBackend(backend, testBitmap)
            val current = _uiState.value.session
            val newResults = current?.results
                ?.filter { it.backend != backend }
                ?.plus(result)
                ?.sortedBy { it.backend.ordinal }
                ?: listOf(result)

            _uiState.update { it.copy(
                session = BenchmarkSession(
                    deviceInfo = _uiState.value.deviceInfo
                        ?: DeviceInfo.collect(getApplication()),
                    modelName = "YOLOv8n (320×320)",
                    results   = newResults
                ),
                isRunning = false
            )}
        }
    }

    fun updateLiveFrame(bitmap: Bitmap) {
        _uiState.update { it.copy(liveFrame = bitmap) }
    }
}
```

```kotlin
                .also { it.setSurfaceProvider(previewView.surfaceProvider) }

            val analysis = ImageAnalysis.Builder()
                .setTargetResolution(Size(640, 480))
                .setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
                .build()
                .also {
                    it.setAnalyzer(cameraExecutor) { proxy ->
                        latestBitmap = proxy.toBitmap()
                        proxy.close()
                    }
                }

            cameraProvider.unbindAll()
            cameraProvider.bindToLifecycle(
                this,
                CameraSelector.DEFAULT_BACK_CAMERA,
                preview, analysis
            )
        }, ContextCompat.getMainExecutor(this))
    }

    private fun setupButtons() {
        btnRunAll.setOnClickListener {
            latestBitmap?.let { viewModel.runAllBenchmarks(it) }
                ?: Toast.makeText(this, "Camera not ready", Toast.LENGTH_SHORT).show()
        }

        btnRunCpu.setOnClickListener {
            latestBitmap?.let { viewModel.runSingleBackend(Backend.CPU, it) }
        }

        btnRunGpu.setOnClickListener {
            latestBitmap?.let { viewModel.runSingleBackend(Backend.GPU, it) }
        }

        btnRunNnapi.setOnClickListener {
            latestBitmap?.let { viewModel.runSingleBackend(Backend.NNAPI, it) }
        }

        btnShare.setOnClickListener { shareReport() }
    }

    private fun observeState() {
        lifecycleScope.launch {
            viewModel.uiState.collect { state ->
                // Device info header
                state.deviceInfo?.let { info ->
                    tvDeviceInfo.text = buildString {
                        appendLine("🔧 ${info.displayName}")
                        appendLine("Android API ${info.androidVersion}  ·  " +
                            "${"%.0f".format(info.totalRamGb)} GB RAM")
                        append(info.npuStatus)
                    }
                }

                // Progress
                progressBar.visibility = if (state.isRunning) View.VISIBLE else View.GONE
                tvStep.text = state.currentStep
                tvStep.visibility = if (state.currentStep.isNotEmpty()) View.VISIBLE else View.GONE

                // Disable buttons during run
                listOf(btnRunAll, btnRunCpu, btnRunGpu, btnRunNnapi).forEach {
                    it.isEnabled = !state.isRunning
                }

                // Bar chart
                state.session?.let { session ->
                    barChart.setResults(session.results.filter { it.supported })
                    btnShare.visibility = View.VISIBLE
                }

                state.error?.let {
                    Toast.makeText(this@BenchmarkActivity, it, Toast.LENGTH_LONG).show()
                }
            }
        }
    }

    private fun shareReport() {
        val session = viewModel.uiState.value.session ?: return
        val report  = session.toMarkdown()

        val intent = Intent(Intent.ACTION_SEND).apply {
            type    = "text/plain"
            putExtra(Intent.EXTRA_SUBJECT, "TFLite Benchmark — ${session.deviceInfo.displayName}")
            putExtra(Intent.EXTRA_TEXT, report)
        }
        startActivity(Intent.createChooser(intent, "Share Benchmark Report"))
    }

    override fun onDestroy() {
        super.onDestroy()
        cameraExecutor.shutdown()
    }
}
```
