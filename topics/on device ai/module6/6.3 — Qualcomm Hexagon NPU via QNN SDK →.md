## 1. Qualcomm AI Stack Overview

Qualcomm powers ~40% of all Android flagship phones. Understanding their AI stack is essential:

```
Qualcomm AI Stack (2025):

┌─────────────────────────────────────────────────────┐
│              Your Android App (Kotlin)              │
├─────────────────────────────────────────────────────┤
│  TFLite NNAPI    │  ONNX Runtime    │  QNN SDK      │
│  Delegate        │  QNN EP          │  (direct)     │
├─────────────────────────────────────────────────────┤
│           QNN (Qualcomm Neural Network)             │
│           Unified inference runtime                 │
├──────────────────┬──────────────────────────────────┤
│  Hexagon NPU     │  Adreno GPU      │  Kryo CPU     │
│  (best for AI)   │  (fast parallel) │  (fallback)   │
└──────────────────┴──────────────────┴───────────────┘

Three ways to access Hexagon:
  1. Via NNAPI delegate (easiest — automatic routing)
  2. Via ONNX Runtime QNN Execution Provider (recommended for ONNX models)
  3. Via QNN SDK directly (maximum control, most complex)
```

## 3. Approach 1 — NNAPI with Hexagon (Easiest)

The simplest way to use Hexagon — just enable NNAPI and Qualcomm's driver handles the routing:

```kotlin
// HexagonViaNnapi.kt
package com.yourname.benchmark

import android.content.Context
import android.util.Log
import org.tensorflow.lite.Interpreter
import org.tensorflow.lite.nnapi.NnApiDelegate
import org.tensorflow.lite.support.common.FileUtil

class HexagonViaNnapi(context: Context, modelFileName: String) {

    companion object {
        private const val TAG = "HexagonNNAPI"
    }

    private var nnApiDelegate: NnApiDelegate? = null
    val interpreter: Interpreter

    init {
        val modelBuffer = FileUtil.loadMappedFile(context, modelFileName)

        val nnApiOptions = NnApiDelegate.Options().apply {
            // Target Hexagon specifically on Qualcomm devices
            // "qti-hta" = Qualcomm High-performance Tensor Accelerator (Hexagon)
            // "qti-dsp" = Qualcomm DSP (Hexagon DSP, more power-efficient)
            // Leave null to let NNAPI choose best available accelerator
            setAcceleratorName(null)   // auto-select

            // Execution preference
            setExecutionPreference(
                NnApiDelegate.Options.EXECUTION_PREFERENCE_FAST_SINGLE_ANSWER
            )

            // Enable FP16 — Hexagon supports it natively for speed
            setAllowFp16(true)

            // Cache compiled model for Hexagon (critical — first compile takes 5–10s)
            setCacheDir(context.cacheDir.absolutePath)
            setModelToken("model_hexagon_v1")   // change when model updates

            // Don't use NNAPI's own CPU — use TFLite's CPU instead (faster)
            setUseNnapiCpu(false)
        }

        try {
            nnApiDelegate = NnApiDelegate(nnApiOptions)
            val options   = Interpreter.Options().addDelegate(nnApiDelegate!!)
            interpreter   = Interpreter(modelBuffer, options)

            Log.d(TAG, "NNAPI delegate created — targeting Hexagon NPU")
            logNnapiInfo()

        } catch (e: Exception) {
            Log.w(TAG, "NNAPI failed: ${e.message} — falling back to CPU")
            nnApiDelegate = null
            interpreter = Interpreter(modelBuffer, Interpreter.Options().setNumThreads(4))
        }
    }

    private fun logNnapiInfo() {
        try {
            val diagInfo = nnApiDelegate?.nnApiDiagnosticData
            Log.d(TAG, "NNAPI diagnostic: $diagInfo")
        } catch (e: Exception) { /* optional */ }
    }

    fun close() {
        interpreter.close()
        nnApiDelegate?.close()
    }
}
```

## 4. Approach 2 — ONNX Runtime QNN Execution Provider (Recommended)

For ONNX models, the QNN Execution Provider gives direct Hexagon access:

```kotlin
// OnnxQnnInference.kt
package com.yourname.benchmark

import ai.onnxruntime.*
import android.content.Context
import android.util.Log
import java.io.File

class OnnxQnnInference(
    private val context: Context,
    private val modelFileName: String
) {
    companion object {
        private const val TAG = "OnnxQNN"
    }

    private val ortEnv: OrtEnvironment = OrtEnvironment.getEnvironment()
    private lateinit var session: OrtSession
    private var usingQnn = false

    fun initialize() {
        val modelBytes = context.assets.open(modelFileName).readBytes()
        val cacheDir   = File(context.cacheDir, "qnn_cache").apply { mkdirs() }

        // Try QNN Execution Provider (Hexagon NPU)
        try {
            val sessionOptions = OrtSession.SessionOptions().apply {
                setIntraOpNumThreads(4)

                // QNN Execution Provider — routes to Hexagon NPU on Qualcomm
                addQnn(mapOf(
                    // Backend: "HTP" = Hexagon Tensor Processor (NPU)
                    //          "DSP" = Hexagon DSP (more power-efficient)
                    //          "GPU" = Qualcomm Adreno GPU
                    //          "CPU" = Qualcomm Kryo CPU
                    "backend_path"    to "libQnnHtp.so",    // HTP = Hexagon NPU
                    "htp_performance_mode" to "burst",       // max performance
                    "qnn_context_cache_enable" to "1",       // cache compiled model
                    "qnn_context_cache_path" to "${cacheDir.absolutePath}/qnn_cache",
                ))
            }

            session  = ortEnv.createSession(modelBytes, sessionOptions)
            usingQnn = true
            Log.d(TAG, "✅ QNN Execution Provider active (Hexagon HTP)")

        } catch (e: OrtException) {
            Log.w(TAG, "QNN EP failed: ${e.message}")

            // Try NNAPI as fallback
            try {
                val fallbackOptions = OrtSession.SessionOptions().apply {
                    setIntraOpNumThreads(4)
                    addNnapi()
                }
                session  = ortEnv.createSession(modelBytes, fallbackOptions)
                usingQnn = false
                Log.d(TAG, "Using NNAPI fallback")

            } catch (e2: OrtException) {
                // CPU fallback
                val cpuOptions = OrtSession.SessionOptions().apply {
                    setIntraOpNumThreads(4)
                }
                session  = ortEnv.createSession(modelBytes, cpuOptions)
                usingQnn = false
                Log.d(TAG, "Using CPU fallback")
            }
        }
    }

    /**
     * Run inference on a float input array.
     * @param inputArray  Flattened float input matching model's input shape
     * @param inputShape  Shape of the input tensor
     * @return FloatArray of output values
     */
    fun runInference(inputArray: FloatArray, inputShape: LongArray): FloatArray {
        val inputName  = session.inputNames.first()
        val outputName = session.outputNames.first()

        val inputTensor = OnnxTensor.createTensor(
            ortEnv,
            java.nio.FloatBuffer.wrap(inputArray),
            inputShape
        )

        val results = session.run(mapOf(inputName to inputTensor))
        val output  = results[outputName]?.value as Array<FloatArray>

        inputTensor.close()
        results.close()

        return output[0]
    }

    fun isUsingQnn(): Boolean = usingQnn

    fun close() {
        session.close()
        ortEnv.close()
    }
}
```

## 5. Approach 3 — QNN SDK Direct Integration

For maximum control, use QNN SDK directly. This requires the Qualcomm AI Engine Direct SDK:

```kotlin
// QnnDirectInference.kt
// Requires: Qualcomm AI Engine Direct SDK AAR
// Download from: https://qpm.qualcomm.com/

package com.yourname.benchmark

import android.content.Context
import android.util.Log

/**
 * Direct QNN SDK usage via JNI.
 * Use this when you need:
 *  - Access to Hexagon-specific optimizations not in NNAPI
 *  - Custom quantization profiles
 *  - Hexagon Debug infrastructure
 *  - Models in DLC format (Qualcomm proprietary)
 *
 * IMPORTANT: QNN SDK requires signing a license with Qualcomm.
 * For most apps, Approach 1 (NNAPI) or 2 (ONNX QNN EP) is sufficient.
 */
class QnnDirectInference(private val context: Context) {

    companion object {
        private const val TAG = "QnnDirect"

        // Load QNN native libraries
        init {
            try {
                System.loadLibrary("QnnHtp")         // Hexagon HTP backend
                System.loadLibrary("QnnHtpNetRunV73") // Hexagon runtime v73+
                System.loadLibrary("QnnSystem")       // QNN system utilities
                Log.d(TAG, "QNN native libraries loaded")
            } catch (e: UnsatisfiedLinkError) {
                Log.w(TAG, "QNN libraries not available: ${e.message}")
            }
        }

        // JNI function declarations
        @JvmStatic external fun qnnInit(modelPath: String): Long
        @JvmStatic external fun qnnRunInference(handle: Long, input: FloatArray): FloatArray
        @JvmStatic external fun qnnClose(handle: Long)
    }

    private var modelHandle: Long = 0

    /**
     * Initialize QNN model from a .bin file compiled with qnn-net-run.
     * The model must be pre-compiled for Hexagon using QNN SDK tools.
     */
    fun initialize(modelBinPath: String): Boolean {
        return try {
            modelHandle = qnnInit(modelBinPath)
            modelHandle != 0L
        } catch (e: UnsatisfiedLinkError) {
            Log.e(TAG, "QNN SDK not linked — use NNAPI or ONNX QNN EP instead")
            false
        }
    }

    fun runInference(input: FloatArray): FloatArray? {
        if (modelHandle == 0L) return null
        return try {
            qnnRunInference(modelHandle, input)
        } catch (e: Exception) {
            Log.e(TAG, "QNN inference failed: ${e.message}")
            null
        }
    }

    fun close() {
        if (modelHandle != 0L) {
            qnnClose(modelHandle)
            modelHandle = 0
        }
    }
}
```

## 6. Converting Models for QNN / Hexagon

To get the best Hexagon performance, convert your model using Qualcomm's tools:

```bash
# Install Qualcomm AI Engine Direct SDK
# Download from: https://qpm.qualcomm.com/

# Convert ONNX model to QNN format (.bin)
qnn-onnx-converter \
  --input_network yolov8n.onnx \
  --output_path yolov8n_qnn \
  --input_dim input "1,3,320,320" \
  --quantization_overrides quant_config.json  # optional INT8

# Compile for Hexagon HTP (generates .serialized.bin)
qnn-context-binary-generator \
  --backend libQnnHtp.so \
  --model yolov8n_qnn.bin \
  --output_dir yolov8n_hexagon \
  --binary_file yolov8n_hexagon

# The output file yolov8n_hexagon.bin is pre-compiled for Hexagon
# Copy to Android assets or push via ADB
```

## 7. Hexagon Performance Tiers

```
Hexagon HTP (High-performance Tensor Processor):
  Best for: FP16 and INT8 neural network inference
  Speed: Maximum throughput
  Power: Medium (active compute)
  Use case: Single inference, burst mode, real-time video

Hexagon DSP (Digital Signal Processor):
  Best for: Always-on tasks (wake word, sensor fusion)
  Speed: Moderate
  Power: Lowest (designed for sustained operation)
  Use case: Background AI, edge detection, audio processing

When to use each:
  Time-critical inference → Hexagon HTP (via "backend_path": "libQnnHtp.so")
  Battery-sensitive always-on → Hexagon DSP (via "backend_path": "libQnnDsp.so")
```

## 8. Benchmark: CPU vs GPU vs Hexagon NPU

Run this on a Qualcomm device to compare all three backends:

```kotlin
// HexagonBenchmark.kt
suspend fun benchmarkAllBackends(
    context: Context,
    modelFileName: String,
    inputData: FloatArray,
    inputShape: LongArray,
    runs: Int = 20
): Map<String, BenchmarkResult> = withContext(Dispatchers.IO) {

    val results = mutableMapOf<String, BenchmarkResult>()

    // 1. CPU (ONNX Runtime)
    results["CPU"] = benchmarkBackend("CPU", runs) {
        val options = OrtSession.SessionOptions().setIntraOpNumThreads(4)
        val session = OrtEnvironment.getEnvironment()
            .createSession(context.assets.open(modelFileName).readBytes(), options)
        session.run(mapOf(session.inputNames.first() to
            OnnxTensor.createTensor(OrtEnvironment.getEnvironment(),
                java.nio.FloatBuffer.wrap(inputData), inputShape)))
        session.close()
    }

    // 2. NNAPI (Hexagon via NNAPI)
    results["NNAPI (Hexagon)"] = benchmarkBackend("NNAPI", runs) {
        val options = OrtSession.SessionOptions().apply {
            setIntraOpNumThreads(4)
            addNnapi()
        }
        val session = OrtEnvironment.getEnvironment()
            .createSession(context.assets.open(modelFileName).readBytes(), options)
        session.run(mapOf(session.inputNames.first() to
            OnnxTensor.createTensor(OrtEnvironment.getEnvironment(),
                java.nio.FloatBuffer.wrap(inputData), inputShape)))
        session.close()
    }

    // 3. QNN EP (Direct Hexagon HTP)
    try {
        results["QNN HTP (Hexagon)"] = benchmarkBackend("QNN", runs) {
            val options = OrtSession.SessionOptions().apply {
                addQnn(mapOf("backend_path" to "libQnnHtp.so"))
            }
            val session = OrtEnvironment.getEnvironment()
                .createSession(context.assets.open(modelFileName).readBytes(), options)
            session.run(mapOf(session.inputNames.first() to
                OnnxTensor.createTensor(OrtEnvironment.getEnvironment(),
                    java.nio.FloatBuffer.wrap(inputData), inputShape)))
            session.close()
        }
    } catch (e: Exception) {
        results["QNN HTP (Hexagon)"] = BenchmarkResult(0f, 0f, "Not available: ${e.message}")
    }

    results
}

data class BenchmarkResult(
    val avgMs: Float,
    val tokPerSec: Float,
    val note: String = ""
)

suspend fun benchmarkBackend(
    name: String,
    runs: Int,
    block: () -> Unit
): BenchmarkResult = withContext(Dispatchers.IO) {
    // Warmup
    repeat(3) { try { block() } catch (e: Exception) { } }

    // Timed runs
    val times = mutableListOf<Long>()
    repeat(runs) {
        val start = System.nanoTime()
        block()
        times.add((System.nanoTime() - start) / 1_000_000)
    }

    val avgMs = times.average().toFloat()
    Log.d("Benchmark", "$name: avg=${avgMs}ms min=${times.min()}ms max=${times.max()}ms")

    BenchmarkResult(avgMs = avgMs, tokPerSec = 1000f / avgMs)
}
```

**Expected results on Snapdragon 8 Gen 3 (MobileNetV2):**
```
CPU (4 threads):     ~85ms
NNAPI (Hexagon):     ~11ms   ← 7.7× faster
QNN HTP (Hexagon):   ~8ms    ← 10.6× faster
```
