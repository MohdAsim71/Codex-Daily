## 1. MediaPipe LLM Inference API Overview

```
Your Kotlin Code
      │
      ▼
MediaPipe LLM Inference API
  ├─ Loads .bin model from device storage
  ├─ Runs on GPU (OpenGL ES) by default
  ├─ Streams tokens via callback as they generate
  └─ Returns full response when done

Supported models (2025):
  ✅ Gemma 2B IT (instruction-tuned)  ← use this
  ✅ Gemma 7B IT                      (flagship only)
  ✅ Phi-2
  ✅ StableLM 3B
  ✅ Falcon RW 1B (CPU only)
```

## 4. GemmaInferenceHelper.kt — Core Implementation

```kotlin
// GemmaInferenceHelper.kt
package com.yourname.chatapp

import android.content.Context
import android.util.Log
import com.google.mediapipe.tasks.genai.llminference.LlmInference
import com.google.mediapipe.tasks.genai.llminference.LlmInference.LlmInferenceOptions
import kotlinx.coroutines.channels.Channel
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.receiveAsFlow

class GemmaInferenceHelper(
    private val context: Context,
    private val modelPath: String,   // full path to .bin file on device
    private val useGpu: Boolean = true
) {
    companion object {
        private const val TAG = "GemmaHelper"

        // Gemma 2B instruction-tuned prompt format
        // <start_of_turn>user\n{message}<end_of_turn>\n<start_of_turn>model\n
        private const val USER_PREFIX  = "<start_of_turn>user\n"
        private const val USER_SUFFIX  = "<end_of_turn>\n<start_of_turn>model\n"
        private const val MODEL_SUFFIX = "<end_of_turn>\n"
    }

    private var llmInference: LlmInference? = null
    private var isInitialized = false

    /**
     * Initialize Gemma. Call once — takes 2–5 seconds on first load.
     * Must be called before any inference.
     */
    fun initialize() {
        try {
            val options = LlmInferenceOptions.builder()
                .setModelPath(modelPath)
                .setMaxTokens(1024)          // max output tokens per response
                .setMaxTopK(40)              // top-K sampling
                .setTemperature(0.8f)        // creativity: 0=deterministic, 1=creative
                .setRandomSeed(42)
                .setPreferredBackend(
                    if (useGpu) LlmInference.Backend.GPU
                    else        LlmInference.Backend.CPU
                )
                .build()

            llmInference = LlmInference.createFromOptions(context, options)
            isInitialized = true
            Log.d(TAG, "Gemma 2B initialized on ${if (useGpu) "GPU" else "CPU"}")

        } catch (e: Exception) {
            Log.e(TAG, "Initialization failed: ${e.message}")
            throw e
        }
    }

    /**
     * Generate a response for a single user message.
     * Returns the full response text (blocking).
     */
    fun generateResponse(userMessage: String): String {
        check(isInitialized) { "Call initialize() before generateResponse()" }

        val prompt = buildPrompt(userMessage)
        Log.d(TAG, "Generating response for: '${userMessage.take(50)}'")

        return llmInference?.generateResponse(prompt)
            ?.trimEnd()
            ?: throw IllegalStateException("LLM not initialized")
    }

    /**
     * Generate a response with streaming — tokens arrive one by one via Flow.
     * Use this for real-time UI updates.
     */
    fun generateResponseStream(userMessage: String): Flow<String> {
        check(isInitialized) { "Call initialize() before generateResponseStream()" }

        val channel = Channel<String>(Channel.UNLIMITED)
        val prompt  = buildPrompt(userMessage)

        llmInference?.generateResponseAsync(
            prompt,
            { partialResult, done ->
                if (!partialResult.isNullOrEmpty()) {
                    channel.trySend(partialResult)
                }
                if (done) {
                    channel.close()
                }
            }
        )

        return channel.receiveAsFlow()
    }

    /**
     * Generate response for a multi-turn conversation.
     * [history] is a list of (role, message) pairs.
     * role: "user" or "model"
     */
    fun generateWithHistory(
        history: List<Pair<String, String>>,
        newUserMessage: String
    ): Flow<String> {
        check(isInitialized) { "Call initialize() before generateWithHistory()" }

        val prompt = buildMultiTurnPrompt(history, newUserMessage)
        Log.d(TAG, "Multi-turn prompt (${history.size} turns + new message)")

        val channel = Channel<String>(Channel.UNLIMITED)

        llmInference?.generateResponseAsync(
            prompt,
            { partialResult, done ->
                if (!partialResult.isNullOrEmpty()) {
                    channel.trySend(partialResult)
                }
                if (done) channel.close()
            }
        )

        return channel.receiveAsFlow()
    }

    /**
     * Build Gemma 2B instruction-tuned prompt format.
     * Single turn: user message only.
     */
    private fun buildPrompt(userMessage: String): String {
        return "$USER_PREFIX$userMessage$USER_SUFFIX"
    }

    /**
     * Build multi-turn conversation prompt.
     * Includes full conversation history in Gemma format.
     */
    private fun buildMultiTurnPrompt(
        history: List<Pair<String, String>>,
        newUserMessage: String
    ): String {
        val sb = StringBuilder()

        // Add conversation history
        for ((role, message) in history) {
            when (role) {
                "user"  -> sb.append(USER_PREFIX).append(message).append(USER_SUFFIX)
                "model" -> sb.append(message).append(MODEL_SUFFIX)
            }
        }

        // Add new user message
        sb.append(USER_PREFIX).append(newUserMessage).append(USER_SUFFIX)

        return sb.toString()
    }

    /**
     * Count approximate tokens in a string.
     * Rough estimate: 1 token ≈ 4 characters for English text.
     */
    fun estimateTokenCount(text: String): Int = text.length / 4

    fun isReady() = isInitialized

    fun close() {
        llmInference?.close()
        llmInference = null
        isInitialized = false
        Log.d(TAG, "Gemma closed — GPU memory released")
    }
}
```

## 5. Simple Single-Turn Usage

```kotlin
// Single-turn example — blocking (use Dispatchers.IO)
lifecycleScope.launch(Dispatchers.IO) {
    val helper = GemmaInferenceHelper(
        context   = this@MainActivity,
        modelPath = "${filesDir.absolutePath}/gemma-2b-it-gpu-int4.bin"
    )

    helper.initialize()  // ~3s on first call

    val response = helper.generateResponse(
        "Summarize this in 2 sentences: On-device AI runs models locally " +
        "on user hardware without sending data to the cloud."
    )

    withContext(Dispatchers.Main) {
        textView.text = response
    }

    helper.close()
}
```

## 7. Model Initialization States

Always handle initialization states — it takes 2–8 seconds:

```kotlin
// ModelState.kt
sealed class ModelState {
    object NotLoaded    : ModelState()
    object Loading      : ModelState()
    object Ready        : ModelState()
    data class Error(val message: String) : ModelState()
}

// In ViewModel
private val _modelState = MutableStateFlow<ModelState>(ModelState.NotLoaded)
val modelState = _modelState.asStateFlow()

fun initializeModel() {
    viewModelScope.launch(Dispatchers.IO) {
        _modelState.value = ModelState.Loading
        try {
            gemmaHelper.initialize()
            _modelState.value = ModelState.Ready
        } catch (e: Exception) {
            _modelState.value = ModelState.Error(e.message ?: "Init failed")
        }
    }
}
```

val SYSTEM_PROMPTS = mapOf(

## 9. GPU vs CPU Inference Comparison

```kotlin
// Benchmark GPU vs CPU
suspend fun benchmarkBackends(testPrompt: String, context: Context, modelPath: String) {

    listOf(true, false).forEach { useGpu ->
        val helper = GemmaInferenceHelper(context, modelPath, useGpu = useGpu)
        helper.initialize()

        val start = System.currentTimeMillis()
        val response = helper.generateResponse(testPrompt)
        val elapsed = System.currentTimeMillis() - start

        val tokenCount = helper.estimateTokenCount(response)
        val tokPerSec  = tokenCount * 1000f / elapsed

        Log.d("Benchmark",
            "${if (useGpu) "GPU" else "CPU"}: " +
            "${elapsed}ms | ~$tokenCount tokens | ${tokPerSec.toInt()} tok/s"
        )

        helper.close()
    }
}
```

**Expected output (Pixel 8 Pro):**
```
GPU: 4200ms | ~85 tokens | 20 tok/s
CPU: 18500ms | ~85 tokens | 4 tok/s   ← GPU is ~5× faster
```

## 10. Common Errors and Fixes

| Error | Cause | Fix |
|---|---|---|
| `FileNotFoundException` | Wrong model path | Use `context.filesDir` not `assets/` — model is too large for assets |
| `OutOfMemoryError` | Device RAM too low | Use smaller model or reduce `maxTokens` |
| `IllegalStateException: GPU not supported` | Old GPU driver | Set `useGpu = false` as fallback |
| `generateResponse` returns empty | Prompt format wrong | Use exact Gemma format with `<start_of_turn>` tags |
| Model loads but hangs | Thermal throttling | Add cooldown between long inference runs |
| First inference slow (~8s) | Shader compilation | Expected — subsequent calls are 2–3× faster |

## 11. Summary

- MediaPipe LLM Inference API is the easiest way to run Gemma 2B on Android
- Model must be stored in device storage (not assets — too large)
- Always initialize on `Dispatchers.IO` — takes 2–8 seconds
- Use `generateResponseAsync` with a channel for streaming UI updates
- GPU backend is ~5× faster than CPU — always try GPU first with fallback
- Gemma 2B uses `<start_of_turn>user/model<end_of_turn>` prompt format
