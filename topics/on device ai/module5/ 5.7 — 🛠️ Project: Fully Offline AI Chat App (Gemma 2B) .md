# 5.7 — 🛠️ Project: Fully Offline AI Chat App (Gemma 2B)

> **Module 5 · Project**  
> 🛠️ Build · ⏱️ ~90 min

---

## 🎯 Project Goal

Build a **production-quality fully offline AI chat Android app** that:

- Runs Gemma 2B INT4 entirely on-device — no internet, no API key
- Streams tokens to the UI in real-time with typewriter effect
- Supports multi-turn conversations with context memory
- Has multiple AI personas (Assistant, Summarizer, Code Helper)
- Includes a RAG mode — ask questions about your personal documents
- Shows real-time performance stats (tok/s, RAM usage)
- Handles memory pressure gracefully with auto-unload
- Downloads the model on first launch with progress UI

## 4. ChatViewModel.kt — Full Implementation

```kotlin
// viewmodel/ChatViewModel.kt
package com.yourname.chatapp.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.yourname.chatapp.memory.LazyLlmManager
import com.yourname.chatapp.memory.MemoryChecker
import com.yourname.chatapp.ml.RagPipeline
import com.yourname.chatapp.model.ChatMessage
import com.yourname.chatapp.model.Persona
import kotlinx.coroutines.*
import kotlinx.coroutines.flow.*

data class ChatUiState(
    val messages: List<ChatMessage>   = emptyList(),
    val isGenerating: Boolean         = false,
    val streamingText: String         = "",
    val currentPersona: Persona       = Persona.ASSISTANT,
    val isRagMode: Boolean            = false,
    val tokensPerSec: Float           = 0f,
    val modelState: String            = "Loading...",
    val availableRamMb: Long          = 0L,
    val error: String?                = null
)

class ChatViewModel(application: Application) : AndroidViewModel(application) {

    private val llmManager = LazyLlmManager(application)
    private val ragPipeline = RagPipeline(
        context   = application,
        llmEngine = null   // injected after LLM loads
    )

    private val _uiState = MutableStateFlow(ChatUiState())
    val uiState: StateFlow<ChatUiState> = _uiState.asStateFlow()

    private var generationJob: Job? = null
    private val history = mutableListOf<Pair<String, String>>()
    private var turnCount = 0

    init {
        loadModel()
        updateMemoryStats()
    }

    // ── Model Loading ───────────────────────────────────

    private fun loadModel() {
        viewModelScope.launch {
            val recommendation = MemoryChecker.recommendModel(getApplication())

            if (recommendation == MemoryChecker.ModelRecommendation.NOT_SUPPORTED) {
                _uiState.update { it.copy(
                    modelState = "❌ Not enough RAM for LLMs on this device",
                    error = "This device needs at least 3 GB free RAM for on-device LLMs."
                )}
                return@launch
            }

            _uiState.update { it.copy(modelState = "Loading Gemma 2B...") }
            llmManager.ensureLoaded()

            llmManager.loadState.collect { state ->
                when (state) {
                    is LazyLlmManager.LoadState.Loading ->
                        _uiState.update { it.copy(modelState = "⏳ Loading model...") }
                    is LazyLlmManager.LoadState.Ready ->
                        _uiState.update { it.copy(modelState = "✅ Gemma 2B · GPU") }
                    is LazyLlmManager.LoadState.Error ->
                        _uiState.update { it.copy(
                            modelState = "❌ ${state.message}",
                            error = state.message
                        )}
                    else -> {}
                }
            }
        }
    }

    // ── Sending Messages ────────────────────────────────

    fun sendMessage(userText: String) {
        if (_uiState.value.isGenerating) return
        if (userText.isBlank()) return

        val engine = llmManager.getEngine() ?: run {
            _uiState.update { it.copy(error = "Model not ready. Please wait.") }
            return
        }

        val userMsg = ChatMessage(role = "user", content = userText)
        history.add("user" to userText)
        turnCount++

        _uiState.update { it.copy(
            messages      = it.messages + userMsg,
            isGenerating  = true,
            streamingText = "",
            error         = null
        )}

        generationJob = viewModelScope.launch {
            val sb        = StringBuilder()
            val startTime = System.currentTimeMillis()
            var tokenCount = 0

            try {
                val tokenFlow = if (_uiState.value.isRagMode) {
                    // RAG mode — ground answer in documents
                    val result = ragPipeline.query(userText)
                    result.answer
                } else {
                    // Normal chat mode
                    engine.generateWithHistory(
                        history     = history.dropLast(1),
                        newUserMessage = userText
                    )
                }

                tokenFlow.collect { token ->
                    sb.append(token)
                    tokenCount++
                    val elapsed = (System.currentTimeMillis() - startTime) / 1000f
                    val tps = if (elapsed > 0) tokenCount / elapsed else 0f

                    _uiState.update { it.copy(
                        streamingText = sb.toString(),
                        tokensPerSec  = tps
                    )}
                }

                val assistantMsg = ChatMessage(
                    role    = "assistant",
                    content = sb.toString()
                )
                history.add("assistant" to sb.toString())

                _uiState.update { it.copy(
                    messages      = it.messages + assistantMsg,
                    isGenerating  = false,
                    streamingText = ""
                )}

                // Reset KV cache every 25 turns to prevent OOM
                if (turnCount >= 25) {
                    engine.close()
                    engine.initialize()
                    turnCount = 0
                    Log.d("Chat", "KV cache reset after 25 turns")
                }

            } catch (e: CancellationException) {
                if (sb.isNotEmpty()) {
                    _uiState.update { it.copy(
                        messages = it.messages + ChatMessage(
                            role = "assistant",
                            content = "${sb.toString()} ✋"
                        ),
                        isGenerating  = false,
                        streamingText = ""
                    )}
                } else {
                    _uiState.update { it.copy(isGenerating = false, streamingText = "") }
                }
            } catch (e: OutOfMemoryError) {
                llmManager.unload()
                _uiState.update { it.copy(
                    isGenerating  = false,
                    streamingText = "",
                    error = "Out of memory. Close other apps and restart."
                )}
            }
        }
    }

    // ── Controls ────────────────────────────────────────

    fun stopGeneration() {
        generationJob?.cancel()
    }

    fun setPersona(persona: Persona) {
        _uiState.update { it.copy(currentPersona = persona) }
    }

    fun setRagMode(enabled: Boolean) {
        _uiState.update { it.copy(isRagMode = enabled) }
    }

    fun clearChat() {
        stopGeneration()
        history.clear()
        turnCount = 0
        llmManager.getEngine()?.close()
        _uiState.update { it.copy(
            messages      = emptyList(),
            streamingText = "",
            isGenerating  = false,
            error         = null
        )}
    }

    // ── RAG Document Indexing ───────────────────────────

    fun indexDocument(title: String, content: String) {
        viewModelScope.launch(Dispatchers.IO) {
            ragPipeline.indexDocument(
                docId    = title,
                text     = content,
                metadata = mapOf("source" to title)
            )
        }
    }

    // ── Memory Stats ────────────────────────────────────

    private fun updateMemoryStats() {
        viewModelScope.launch {
            while (true) {
                val info = MemoryChecker.getMemoryInfo(getApplication())
                _uiState.update { it.copy(availableRamMb = info.availableRamMb) }
                delay(5000)   // update every 5 seconds
            }
        }
    }

    override fun onCleared() {
        super.onCleared()
        ragPipeline.close()
        llmManager.destroy()
    }
}
```

## 5. DownloadViewModel.kt — Model Download with Progress

```kotlin
// viewmodel/DownloadViewModel.kt
package com.yourname.chatapp.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.yourname.chatapp.model.ModelDownloadState
import kotlinx.coroutines.*
import kotlinx.coroutines.flow.*
import okhttp3.*
import java.io.File
import java.io.FileOutputStream

class DownloadViewModel(application: Application) : AndroidViewModel(application) {

    // In production, host model on your CDN or use HuggingFace Hub
    private val MODEL_URL = "https://your-cdn.com/gemma-2b-it-gpu-int4.bin"

    private val _downloadState = MutableStateFlow<ModelDownloadState>(
        ModelDownloadState.Checking
    )
    val downloadState: StateFlow<ModelDownloadState> = _downloadState.asStateFlow()

    val modelFile: File get() =
        File(getApplication<Application>().filesDir, "gemma-2b-it-gpu-int4.bin")

    init { checkModelExists() }

    private fun checkModelExists() {
        if (modelFile.exists() && modelFile.length() > 100_000_000L) {
            // Model already downloaded (> 100 MB = plausible)
            _downloadState.value = ModelDownloadState.AlreadyExists
        } else {
            _downloadState.value = ModelDownloadState.Checking
        }
    }

    fun startDownload() {
        viewModelScope.launch(Dispatchers.IO) {
            try {
                val client = OkHttpClient()
                val request = Request.Builder().url(MODEL_URL).build()
                val response = client.newCall(request).execute()

                if (!response.isSuccessful) {
                    _downloadState.value = ModelDownloadState.Error(
                        "Download failed: HTTP ${response.code}"
                    )
                    return@launch
                }

                val totalBytes = response.body?.contentLength() ?: -1L
                val totalMb    = totalBytes / (1024f * 1024f)

                FileOutputStream(modelFile).use { output ->
                    response.body?.byteStream()?.use { input ->
                        val buffer     = ByteArray(8192)
                        var downloaded = 0L
                        var lastUpdate = System.currentTimeMillis()
                        var lastBytes  = 0L
                        var bytesRead: Int

                        while (input.read(buffer).also { bytesRead = it } != -1) {
                            output.write(buffer, 0, bytesRead)
                            downloaded += bytesRead

                            val now = System.currentTimeMillis()
                            if (now - lastUpdate >= 500) {   // update every 500ms
                                val speedMbps = ((downloaded - lastBytes) * 2f) /
                                    (1024f * 1024f)  // bytes/500ms * 2 = bytes/sec → MB/s
                                val progress  = if (totalBytes > 0)
                                    ((downloaded * 100) / totalBytes).toInt() else 0

                                _downloadState.value = ModelDownloadState.Downloading(
                                    progressPercent = progress,
                                    downloadedMb    = downloaded / (1024f * 1024f),
                                    totalMb         = totalMb,
                                    speedMbps       = speedMbps
                                )

                                lastUpdate = now
                                lastBytes  = downloaded
                            }
                        }
                    }
                }

                _downloadState.value = ModelDownloadState.DownloadComplete

            } catch (e: Exception) {
                modelFile.delete()   // clean up partial download
                _downloadState.value = ModelDownloadState.Error(
                    "Download failed: ${e.message}"
                )
            }
        }
    }
}
```

## 7. MainActivity.kt — Download Screen

```kotlin
// ui/MainActivity.kt
package com.yourname.chatapp.ui

import android.content.Intent
import android.os.Bundle
import android.view.View
import android.widget.*
import androidx.activity.viewModels
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import com.google.android.material.button.MaterialButton
import com.google.android.material.progressindicator.LinearProgressIndicator
import com.yourname.chatapp.R
import com.yourname.chatapp.model.ModelDownloadState
import com.yourname.chatapp.viewmodel.DownloadViewModel
import kotlinx.coroutines.launch

class MainActivity : AppCompatActivity() {

    private val viewModel: DownloadViewModel by viewModels()

    private lateinit var tvStatus: TextView
    private lateinit var tvDownloadDetails: TextView
    private lateinit var progressBar: LinearProgressIndicator
    private lateinit var btnAction: MaterialButton

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        tvStatus         = findViewById(R.id.tvStatus)
        tvDownloadDetails = findViewById(R.id.tvDownloadDetails)
        progressBar      = findViewById(R.id.progressBar)
        btnAction        = findViewById(R.id.btnAction)

        observeDownloadState()
    }

    private fun observeDownloadState() {
        lifecycleScope.launch {
            viewModel.downloadState.collect { state ->
                when (state) {
                    is ModelDownloadState.Checking -> {
                        tvStatus.text = "Checking for model..."
                        progressBar.visibility = View.GONE
                        btnAction.visibility   = View.GONE
                    }

                    is ModelDownloadState.AlreadyExists -> {
                        // Model ready — go straight to chat
                        navigateToChat()
                    }

                    is ModelDownloadState.Downloading -> {
                        tvStatus.text = "Downloading Gemma 2B..."
                        progressBar.visibility    = View.VISIBLE
                        progressBar.progress      = state.progressPercent
                        tvDownloadDetails.visibility = View.VISIBLE
                        tvDownloadDetails.text    =
                            "${state.downloadedMb.toInt()} / ${state.totalMb.toInt()} MB  " +
                            "·  ${"%.1f".format(state.speedMbps)} MB/s  " +
                            "·  ${state.progressPercent}%"
                        btnAction.visibility = View.GONE
                    }

                    is ModelDownloadState.DownloadComplete -> {
                        tvStatus.text = "✅ Model ready!"
                        progressBar.visibility = View.GONE
                        delay(1000)
                        navigateToChat()
                    }

                    is ModelDownloadState.Error -> {
                        tvStatus.text = "❌ ${state.message}"
                        progressBar.visibility = View.GONE
                        btnAction.text       = "Retry Download"
                        btnAction.visibility = View.VISIBLE
                        btnAction.setOnClickListener { viewModel.startDownload() }
                    }
                }
            }
        }

        // Show download button after checking
        btnAction.setOnClickListener { viewModel.startDownload() }
    }

    private fun navigateToChat() {
        startActivity(Intent(this, ChatActivity::class.java))
        finish()
    }
}
```

## 9. ChatActivity.kt — Full Implementation

```kotlin
// ui/ChatActivity.kt
package com.yourname.chatapp.ui

import android.os.Bundle
import android.view.View
import android.widget.*
import androidx.activity.viewModels
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.*
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.google.android.material.chip.Chip
import com.yourname.chatapp.R
import com.yourname.chatapp.model.ChatMessage
import com.yourname.chatapp.model.Persona
import com.yourname.chatapp.viewmodel.ChatViewModel
import kotlinx.coroutines.launch

class ChatActivity : AppCompatActivity() {

    private val viewModel: ChatViewModel by viewModels()

    private lateinit var recyclerView: RecyclerView
    private lateinit var adapter: ChatAdapter
    private lateinit var etInput: EditText
    private lateinit var btnSend: ImageButton
    private lateinit var btnStop: ImageButton
    private lateinit var typingIndicator: TypingIndicatorView
    private lateinit var tvModelStatus: TextView
    private lateinit var tvStats: TextView
    private lateinit var spinnerPersona: Spinner
    private lateinit var chipRag: Chip
    private lateinit var btnClear: ImageButton

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_chat)

        bindViews()
        setupRecyclerView()
        setupPersonaSpinner()
        setupListeners()
        observeUiState()
    }

    private fun bindViews() {
        recyclerView    = findViewById(R.id.recyclerView)
        etInput         = findViewById(R.id.etInput)
        btnSend         = findViewById(R.id.btnSend)
        btnStop         = findViewById(R.id.btnStop)
        typingIndicator = findViewById(R.id.typingIndicator)
        tvModelStatus   = findViewById(R.id.tvModelStatus)
        tvStats         = findViewById(R.id.tvStats)
        spinnerPersona  = findViewById(R.id.spinnerPersona)
        chipRag         = findViewById(R.id.chipRag)
        btnClear        = findViewById(R.id.btnClear)
    }

    private fun setupRecyclerView() {
        adapter = ChatAdapter()
        recyclerView.apply {
            layoutManager = LinearLayoutManager(this@ChatActivity).apply {
                stackFromEnd = true
            }
            adapter = this@ChatActivity.adapter
        }
    }

    private fun setupPersonaSpinner() {
        val personas = Persona.values()
        val names    = personas.map { "${it.emoji} ${it.displayName}" }
        spinnerPersona.adapter = ArrayAdapter(
            this, android.R.layout.simple_spinner_item, names
        ).apply {
            setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)
        }
        spinnerPersona.onItemSelectedListener = object : AdapterView.OnItemSelectedListener {
            override fun onItemSelected(p: AdapterView<*>?, v: View?, pos: Int, id: Long) {
                viewModel.setPersona(personas[pos])
            }
            override fun onNothingSelected(p: AdapterView<*>?) {}
        }
    }

    private fun setupListeners() {
        btnSend.setOnClickListener {
            val text = etInput.text.toString().trim()
            if (text.isNotEmpty()) {
                viewModel.sendMessage(text)
                etInput.text.clear()
            }
        }

        btnStop.setOnClickListener { viewModel.stopGeneration() }

        btnClear.setOnClickListener { viewModel.clearChat() }

        chipRag.setOnCheckedChangeListener { _, checked ->
            viewModel.setRagMode(checked)
        }
    }

    private fun observeUiState() {
        lifecycleScope.launch {
            viewModel.uiState
                .flowWithLifecycle(lifecycle, Lifecycle.State.STARTED)
                .collect { state -> renderState(state) }
        }
    }

    private fun renderState(state: com.yourname.chatapp.viewmodel.ChatUiState) {
        // Model status bar
        tvModelStatus.text = buildString {
            append(state.modelState)
            if (state.availableRamMb > 0)
                append("  ·  ${state.availableRamMb} MB free")
        }

        // Messages + streaming
        val displayMessages = state.messages.toMutableList()
        if (state.isGenerating && state.streamingText.isNotEmpty()) {
            displayMessages.add(ChatMessage(
                id          = -1L,
                role        = "assistant",
                content     = state.streamingText,
                isStreaming = true
            ))
        }
        adapter.submitList(displayMessages.toList()) {
            if (displayMessages.isNotEmpty()) {
                recyclerView.smoothScrollToPosition(displayMessages.size - 1)
            }
        }

        // Typing indicator (waiting for first token)
        if (state.isGenerating && state.streamingText.isEmpty()) {
            typingIndicator.startAnimation()
        } else {
            typingIndicator.stopAnimation()
        }

        // Send / Stop button swap
        btnSend.visibility = if (state.isGenerating) View.GONE  else View.VISIBLE
        btnStop.visibility = if (state.isGenerating) View.VISIBLE else View.GONE
        etInput.isEnabled  = !state.isGenerating

        // Token speed
        tvStats.text = if (state.tokensPerSec > 0)
            "${"%.1f".format(state.tokensPerSec)} tok/s" else ""

        // Errors
        state.error?.let {
            Toast.makeText(this, it, Toast.LENGTH_LONG).show()
        }
    }
}
```
