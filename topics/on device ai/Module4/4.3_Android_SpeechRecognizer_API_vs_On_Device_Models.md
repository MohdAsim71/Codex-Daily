## 1. Android SpeechRecognizer API — How It Works

```
User speaks
    │
    ▼
SpeechRecognizer (Android system service)
    │
    ├─ Captures mic audio
    ├─ Detects speech start/end automatically
    ├─ Streams audio to recognition service
    │       (Google by default, configurable)
    └─ Returns results via RecognitionListener callbacks

Key: it manages mic, VAD (voice activity detection), and streaming for you.
```

## 2. Complete SpeechRecognizer Implementation

```kotlin
// SpeechRecognizerHelper.kt
package com.yourname.voicenotes

import android.content.Context
import android.content.Intent
import android.os.Bundle
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import android.util.Log
import java.util.Locale

class SpeechRecognizerHelper(
    private val context: Context,
    private val onPartialResult: (String) -> Unit = {},   // word-by-word updates
    private val onFinalResult:   (String) -> Unit,        // complete transcript
    private val onError:         (String) -> Unit,
    private val onStateChange:   (RecognizerState) -> Unit = {}
) {
    enum class RecognizerState { IDLE, LISTENING, PROCESSING }

    companion object {
        private const val TAG = "SpeechRecognizer"
    }

    private var speechRecognizer: SpeechRecognizer? = null

    init {
        // Always check availability first — not all devices have it
        if (!SpeechRecognizer.isRecognitionAvailable(context)) {
            onError("Speech recognition not available on this device")
        }
    }

    fun startListening(locale: Locale = Locale.ENGLISH) {
        // Create fresh instance — SpeechRecognizer is not reusable after error
        speechRecognizer?.destroy()
        speechRecognizer = SpeechRecognizer.createSpeechRecognizer(context)
        speechRecognizer?.setRecognitionListener(buildListener())

        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
            // Language model: FREE_FORM for natural speech, WEB_SEARCH for queries
            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL,
                RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)

            // Language
            putExtra(RecognizerIntent.EXTRA_LANGUAGE, locale.toLanguageTag())
            putExtra(RecognizerIntent.EXTRA_LANGUAGE_PREFERENCE, locale.toLanguageTag())

            // Return partial results as user speaks (enables word-by-word updates)
            putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)

            // Max number of recognition hypotheses to return
            putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 3)

            // Don't show system UI dialog (handle mic in your own UI)
            putExtra(RecognizerIntent.EXTRA_CALLING_PACKAGE, context.packageName)

            // Speech timeout settings
            putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_COMPLETE_SILENCE_LENGTH_MILLIS, 1500L)
            putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_POSSIBLY_COMPLETE_SILENCE_LENGTH_MILLIS, 1000L)
        }

        onStateChange(RecognizerState.LISTENING)
        speechRecognizer?.startListening(intent)
        Log.d(TAG, "Started listening [${locale.toLanguageTag()}]")
    }

    fun stopListening() {
        speechRecognizer?.stopListening()
        onStateChange(RecognizerState.PROCESSING)
    }

    fun cancel() {
        speechRecognizer?.cancel()
        onStateChange(RecognizerState.IDLE)
    }

    private fun buildListener() = object : RecognitionListener {

        // Called when audio recording begins
        override fun onReadyForSpeech(params: Bundle?) {
            Log.d(TAG, "Ready for speech")
        }

        // Called when speech is detected (mic icon should animate)
        override fun onBeginningOfSpeech() {
            Log.d(TAG, "Speech started")
        }

        // Called continuously with audio level (0–10) — use for visualizer
        override fun onRmsChanged(rmsdB: Float) {
            // rmsdB range: 0 to ~10
            // Use this to animate a mic level indicator in your UI
        }

        // Raw audio buffer (rarely needed)
        override fun onBufferReceived(buffer: ByteArray?) { }

        // Speech ended — processing begins
        override fun onEndOfSpeech() {
            Log.d(TAG, "Speech ended, processing...")
            onStateChange(RecognizerState.PROCESSING)
        }

        // Partial results — called while user is still speaking
        override fun onPartialResults(partialResults: Bundle?) {
            val partial = partialResults
                ?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                ?.firstOrNull() ?: return
            Log.d(TAG, "Partial: $partial")
            onPartialResult(partial)
        }

        // ← Final result — this is the main callback you care about
        override fun onResults(results: Bundle?) {
            val texts = results
                ?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                ?: run {
                    onError("No recognition results")
                    return
                }

            val confidences = results
                .getFloatArray(SpeechRecognizer.CONFIDENCE_SCORES)

            // Log all hypotheses with confidence
            texts.forEachIndexed { i, text ->
                val conf = confidences?.getOrNull(i) ?: -1f
                Log.d(TAG, "Hypothesis[$i]: '$text' (conf=${"%.2f".format(conf)})")
            }

            // Deliver best result (index 0 = highest confidence)
            onFinalResult(texts.first())
            onStateChange(RecognizerState.IDLE)
        }

        // Error callback — handle every error code
        override fun onError(error: Int) {
            val message = when (error) {
                SpeechRecognizer.ERROR_AUDIO              -> "Audio recording error"
                SpeechRecognizer.ERROR_CLIENT             -> "Client-side error"
                SpeechRecognizer.ERROR_INSUFFICIENT_PERMISSIONS -> "Missing RECORD_AUDIO permission"
                SpeechRecognizer.ERROR_NETWORK            -> "Network error — check internet"
                SpeechRecognizer.ERROR_NETWORK_TIMEOUT    -> "Network timeout"
                SpeechRecognizer.ERROR_NO_MATCH           -> "No speech recognized"
                SpeechRecognizer.ERROR_RECOGNIZER_BUSY    -> "Recognizer busy — wait and retry"
                SpeechRecognizer.ERROR_SERVER             -> "Server error"
                SpeechRecognizer.ERROR_SPEECH_TIMEOUT     -> "No speech detected"
                SpeechRecognizer.ERROR_TOO_MANY_REQUESTS  -> "Too many requests"
                SpeechRecognizer.ERROR_LANGUAGE_NOT_SUPPORTED -> "Language not supported"
                SpeechRecognizer.ERROR_LANGUAGE_UNAVAILABLE   -> "Language model unavailable"
                else -> "Unknown error: $error"
            }
            Log.e(TAG, "Error: $message")
            onError(message)
            onStateChange(RecognizerState.IDLE)
        }

        // Called when recognition service is destroyed
        override fun onEvent(eventType: Int, params: Bundle?) { }
    }

    fun destroy() {
        speechRecognizer?.destroy()
        speechRecognizer = null
    }
}
```

## 3. Usage — Voice Command Handler

```kotlin
// In your Activity
private lateinit var sttHelper: SpeechRecognizerHelper
private lateinit var tvTranscript: TextView
private lateinit var tvPartial: TextView

sttHelper = SpeechRecognizerHelper(
    context = this,
    onPartialResult = { partial ->
        tvPartial.text = partial   // live preview as user speaks
    },
    onFinalResult = { text ->
        tvTranscript.text = text
        tvPartial.text = ""
        processVoiceCommand(text)
    },
    onError = { error ->
        Toast.makeText(this, error, Toast.LENGTH_SHORT).show()
    },
    onStateChange = { state ->
        when (state) {
            LISTENING   -> btnMic.text = "🔴 Listening..."
            PROCESSING  -> btnMic.text = "⏳ Processing..."
            IDLE        -> btnMic.text = "🎙️ Speak"
        }
    }
)

btnMic.setOnClickListener { sttHelper.startListening() }

override fun onDestroy() {
    super.onDestroy()
    sttHelper.destroy()
}
```

## 4. Side-by-Side Comparison

| Feature | SpeechRecognizer API | Whisper TFLite |
|---|---|---|
| **Internet required** | ✅ Yes (usually) | ❌ No |
| **Setup complexity** | Low | Medium |
| **Accuracy** | ⭐⭐⭐⭐⭐ | ⭐⭐⭐⭐ (tiny), ⭐⭐⭐⭐⭐ (base) |
| **Streaming / partial** | ✅ Built-in | ❌ Batch only |
| **Voice Activity Detection** | ✅ Automatic | ❌ Manual (you control recording) |
| **Language support** | 100+ | 99 languages |
| **Audio privacy** | ❌ Sent to cloud | ✅ Stays on device |
| **Latency** | 200ms–1s | 500ms–5s (clip-based) |
| **Cost** | Free (Google quota) | Free (no API) |
| **Works offline** | ❌ Unreliable | ✅ Always |
| **Long-form (>30s)** | ✅ | ❌ (30s window) |
| **Noisy environments** | ⭐⭐⭐ | ⭐⭐⭐⭐ |

## 5. Unified STT Interface — Swap Backends Without Changing UI

Good architecture lets you switch STT backends without touching your ViewModel or UI:

```kotlin
// SttProvider.kt
package com.yourname.voicenotes

interface SttProvider {
    fun startListening()
    fun stopListening()
    fun destroy()
    val isAvailable: Boolean
}

// SpeechRecognizer backend
class OnlineSttProvider(
    context: Context,
    onResult: (String) -> Unit,
    onError: (String) -> Unit
) : SttProvider {
    private val helper = SpeechRecognizerHelper(
        context = context,
        onFinalResult = onResult,
        onError = onError
    )
    override fun startListening() = helper.startListening()
    override fun stopListening()  = helper.stopListening()
    override fun destroy()        = helper.destroy()
    override val isAvailable      = SpeechRecognizer.isRecognitionAvailable(context)
}

// Whisper TFLite backend
class OfflineSttProvider(
    context: Context,
    onResult: (String) -> Unit,
    onError: (String) -> Unit
) : SttProvider {
    private val transcriber = WhisperTranscriber(context)
    private val recorder    = AudioRecorder(context)
    private val scope       = CoroutineScope(Dispatchers.IO + SupervisorJob())

    override fun startListening() {
        scope.launch {
            try {
                val audio  = recorder.record(30)
                val result = transcriber.transcribe(audio)
                withContext(Dispatchers.Main) { onResult(result.text) }
            } catch (e: Exception) {
                withContext(Dispatchers.Main) { onError(e.message ?: "Error") }
            }
        }
    }
    override fun stopListening()  = recorder.stopRecording()
    override fun destroy()        { transcriber.close(); scope.cancel() }
    override val isAvailable      = true
}

// Factory — pick the right provider
fun createSttProvider(
    context: Context,
    preferOffline: Boolean,
    onResult: (String) -> Unit,
    onError:  (String) -> Unit
): SttProvider {
    return if (preferOffline || !isNetworkAvailable(context)) {
        OfflineSttProvider(context, onResult, onError)
    } else {
        OnlineSttProvider(context, onResult, onError)
    }
}
```

## 6. Common Pitfalls and Fixes

| Problem | Cause | Fix |
|---|---|---|
| `ERROR_RECOGNIZER_BUSY` | Previous session not destroyed | Call `destroy()` before creating new instance |
| `ERROR_NO_MATCH` repeatedly | Background noise too high | Lower `EXTRA_SPEECH_INPUT_COMPLETE_SILENCE_LENGTH_MILLIS` |
| App crashes on config change | SpeechRecognizer held in Activity | Move to ViewModel, use `onDestroy()` cleanup |
| Partial results not appearing | `EXTRA_PARTIAL_RESULTS` not set | Add `putExtra(EXTRA_PARTIAL_RESULTS, true)` |
| Works on some devices only | Vendor stripped recognition service | Check `isRecognitionAvailable()` and show fallback |

---

## 7. Summary

- `SpeechRecognizer` is the fastest to implement but sends audio to Google's servers
- Use it for: internet-available apps where quality > privacy
- Use Whisper TFLite for: offline apps, sensitive audio, 99-language support
- Always check `isRecognitionAvailable()` — some custom ROMs strip the service
- Build a `SttProvider` interface so you can swap backends without UI changes
- `onRmsChanged` → mic level visualizer; `onPartialResults` → live preview text
