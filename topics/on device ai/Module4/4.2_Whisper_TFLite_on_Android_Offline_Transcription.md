## 1. How Whisper Works

```
Audio (PCM Float32, 16kHz mono)
        │
        ▼
Mel Spectrogram Encoder
(Convert waveform → 80-channel mel spectrogram, 30-second window)
        │
        ▼
Transformer Encoder
(Understand audio features)
        │
        ▼
Transformer Decoder
(Autoregressively generate text tokens)
        │
        ▼
Transcription text
```

The TFLite version of Whisper merges encoder + decoder into one model for simplicity, at the cost of single-pass (non-streaming) transcription.

## 2. Get the Whisper TFLite Model (Python)

Run this once on your PC to export the model:

```python
# export_whisper_tflite.py
# pip install openai-whisper tensorflow

import whisper
import torch
import numpy as np

def export_whisper_to_tflite(model_size="tiny", output_path="whisper_tiny.tflite"):
    """
    Export Whisper to TFLite via PyTorch → ONNX → TF → TFLite pipeline.
    For a faster start, download pre-exported models from:
    https://github.com/usefulsensors/openai-whisper-tflite
    """
    print(f"Loading Whisper {model_size}...")
    model = whisper.load_model(model_size)
    model.eval()

    # Create sample mel input (80 channels × 3000 time frames = 30 seconds)
    sample_mel = torch.zeros(1, 80, 3000)

    # Export encoder only to TorchScript first, then ONNX
    with torch.no_grad():
        traced = torch.jit.trace(model.encoder, sample_mel)

    torch.jit.save(traced, f"whisper_{model_size}_encoder.pt")
    print(f"Encoder saved.")
    print("Note: Full TFLite export requires onnx-tf pipeline.")
    print("Recommended: download pre-built from usefulsensors/openai-whisper-tflite")

# Alternative: use the pre-built TFLite models
print("""
Pre-built Whisper TFLite models (recommended):
  Tiny:  https://huggingface.co/openai/whisper-tiny/tree/main (search for .tflite)
  Base:  whisper.tflite from usefulsensors GitHub

Place in: app/src/main/assets/whisper_tiny.tflite
""")
```

## 5. AudioRecorder.kt — Capture Mic Input

```kotlin
// AudioRecorder.kt
package com.yourname.voicenotes

import android.Manifest
import android.content.Context
import android.media.AudioFormat
import android.media.AudioRecord
import android.media.MediaRecorder
import android.util.Log
import androidx.annotation.RequiresPermission
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext

class AudioRecorder(private val context: Context) {

    companion object {
        private const val TAG            = "AudioRecorder"
        const val SAMPLE_RATE            = 16000   // Whisper requires 16kHz
        const val MAX_DURATION_SECONDS   = 30      // Whisper max window
        const val CHANNEL_CONFIG         = AudioFormat.CHANNEL_IN_MONO
        const val AUDIO_FORMAT           = AudioFormat.ENCODING_PCM_16BIT
    }

    private var audioRecord: AudioRecord? = null
    private var isRecording = false

    /**
     * Record up to [durationSeconds] seconds from the microphone.
     * Returns a Float32 PCM array normalized to [-1.0, 1.0].
     * Requires RECORD_AUDIO permission.
     */
    @RequiresPermission(Manifest.permission.RECORD_AUDIO)
    suspend fun record(durationSeconds: Int = MAX_DURATION_SECONDS): FloatArray =
        withContext(Dispatchers.IO) {

            val minBufferSize = AudioRecord.getMinBufferSize(
                SAMPLE_RATE, CHANNEL_CONFIG, AUDIO_FORMAT
            )
            val totalSamples = SAMPLE_RATE * durationSeconds

            audioRecord = AudioRecord(
                MediaRecorder.AudioSource.MIC,
                SAMPLE_RATE,
                CHANNEL_CONFIG,
                AUDIO_FORMAT,
                minBufferSize * 4   // larger buffer = fewer underruns
            )

            val shortBuffer = ShortArray(totalSamples)
            isRecording = true

            Log.d(TAG, "Recording started: ${durationSeconds}s @ ${SAMPLE_RATE}Hz")
            audioRecord?.startRecording()

            var samplesRead = 0
            while (isRecording && samplesRead < totalSamples) {
                val remaining = totalSamples - samplesRead
                val read = audioRecord?.read(shortBuffer, samplesRead, remaining) ?: 0
                if (read > 0) samplesRead += read
            }

            audioRecord?.stop()
            audioRecord?.release()
            audioRecord = null

            Log.d(TAG, "Recording complete: $samplesRead samples")

            // Convert Int16 PCM → Float32 [-1.0, 1.0]
            FloatArray(samplesRead) { i ->
                shortBuffer[i].toFloat() / 32768.0f
            }
        }

    /** Stop recording early (e.g. user taps Stop button) */
    fun stopRecording() {
        isRecording = false
    }

    val isCurrentlyRecording get() = isRecording
}
```

## 6. MelSpectrogram.kt — Audio Preprocessing

Whisper's encoder expects a mel spectrogram, not raw PCM. We compute it in Kotlin:

```kotlin
// MelSpectrogram.kt
package com.yourname.voicenotes

import kotlin.math.*

/**
 * Computes an 80-channel log mel spectrogram from a Float32 PCM array.
 * This matches Whisper's audio preprocessing exactly.
 */
object MelSpectrogram {

    private const val N_FFT     = 400    // FFT window size
    private const val HOP_LEN   = 160    // hop length (10ms at 16kHz)
    private const val N_MELS    = 80     // mel filter banks
    private const val SR        = 16000  // sample rate

    /**
     * Compute mel spectrogram.
     * @param audio Float32 PCM array at 16kHz, values in [-1, 1]
     * @return FloatArray of shape [N_MELS × N_FRAMES] flattened, then [1 × N_MELS × 3000]
     */
    fun compute(audio: FloatArray): Array<Array<FloatArray>> {
        // Pad audio to exactly 30 seconds (480,000 samples)
        val targetLen = 30 * SR
        val padded = FloatArray(targetLen)
        audio.copyInto(padded, endIndex = minOf(audio.size, targetLen))

        val nFrames = (targetLen - N_FFT) / HOP_LEN + 1

        // Compute STFT magnitude
        val magnitudes = computeStft(padded, nFrames)

        // Apply mel filterbank
        val melFilters = buildMelFilters()
        val melSpec = applyMelFilters(magnitudes, melFilters, nFrames)

        // Log compression + normalize
        val logMelSpec = Array(N_MELS) { mel ->
            FloatArray(3000) { frame ->
                if (frame < nFrames) {
                    val logVal = ln(melSpec[mel][frame].coerceAtLeast(1e-10f))
                    logVal
                } else 0f
            }
        }

        // Normalize to [-1, 1] range (Whisper standard)
        val maxVal = logMelSpec.flatMap { it.toList() }.max().coerceAtLeast(1e-8f)
        logMelSpec.forEach { row ->
            for (i in row.indices) {
                row[i] = ((row[i].coerceAtLeast(maxVal - 8f)) / 4f) + 1f
            }
        }

        // Return as [1, N_MELS, 3000] tensor
        return Array(1) { logMelSpec }
    }

    private fun computeStft(audio: FloatArray, nFrames: Int): Array<FloatArray> {
        val magnitudes = Array(N_FFT / 2 + 1) { FloatArray(nFrames) }
        val window = FloatArray(N_FFT) { i ->
            (0.5f * (1 - cos(2 * PI.toFloat() * i / (N_FFT - 1))))  // Hann window
        }

        for (frame in 0 until nFrames) {
            val start = frame * HOP_LEN
            val real  = FloatArray(N_FFT)
            val imag  = FloatArray(N_FFT)

            for (i in 0 until N_FFT) {
                real[i] = if (start + i < audio.size) audio[start + i] * window[i] else 0f
            }

            fft(real, imag)

            for (bin in 0..N_FFT / 2) {
                magnitudes[bin][frame] = sqrt(real[bin].pow(2) + imag[bin].pow(2))
            }
        }
        return magnitudes
    }

    /** Simple radix-2 Cooley–Tukey FFT */
    private fun fft(real: FloatArray, imag: FloatArray) {
        val n = real.size
        if (n <= 1) return

        // Bit-reversal permutation
        var j = 0
        for (i in 1 until n) {
            var bit = n shr 1
            while (j and bit != 0) { j = j xor bit; bit = bit shr 1 }
            j = j xor bit
            if (i < j) {
                real[i] = real[j].also { real[j] = real[i] }
                imag[i] = imag[j].also { imag[j] = imag[i] }
            }
        }

        var len = 2
        while (len <= n) {
            val ang = -2 * PI / len
            val wReal = cos(ang).toFloat()
            val wImag = sin(ang).toFloat()
            var i = 0
            while (i < n) {
                var curReal = 1f; var curImag = 0f
                for (k in 0 until len / 2) {
                    val tR = curReal * real[i+k+len/2] - curImag * imag[i+k+len/2]
                    val tI = curReal * imag[i+k+len/2] + curImag * real[i+k+len/2]
                    real[i+k+len/2] = real[i+k] - tR
                    imag[i+k+len/2] = imag[i+k] - tI
                    real[i+k] += tR; imag[i+k] += tI
                    val newReal = curReal * wReal - curImag * wImag
                    curImag = curReal * wImag + curImag * wReal; curReal = newReal
                }
                i += len
            }
            len = len shl 1
        }
    }

    private fun buildMelFilters(): Array<FloatArray> {
        val filters = Array(N_MELS) { FloatArray(N_FFT / 2 + 1) }
        val fMin = 0.0; val fMax = SR / 2.0
        val melMin = hzToMel(fMin); val melMax = hzToMel(fMax)
        val melPoints = DoubleArray(N_MELS + 2) { i -> melMin + i * (melMax - melMin) / (N_MELS + 1) }
        val hzPoints  = DoubleArray(N_MELS + 2) { i -> melToHz(melPoints[i]) }
        val binPoints = IntArray(N_MELS + 2) { i -> ((hzPoints[i] / (SR / 2.0)) * (N_FFT / 2)).roundToInt() }

        for (m in 0 until N_MELS) {
            for (k in binPoints[m]..binPoints[m + 2]) {
                filters[m][k] = when {
                    k < binPoints[m]     -> 0f
                    k <= binPoints[m+1]  -> ((k - binPoints[m]).toFloat() / (binPoints[m+1] - binPoints[m]))
                    k <= binPoints[m+2]  -> ((binPoints[m+2] - k).toFloat() / (binPoints[m+2] - binPoints[m+1]))
                    else                 -> 0f
                }
            }
        }
        return filters
    }

    private fun applyMelFilters(
        magnitudes: Array<FloatArray>,
        filters: Array<FloatArray>,
        nFrames: Int
    ): Array<FloatArray> {
        return Array(N_MELS) { mel ->
            FloatArray(nFrames) { frame ->
                var sum = 0f
                for (bin in filters[mel].indices) {
                    sum += filters[mel][bin] * magnitudes[bin][frame]
                }
                sum
            }
        }
    }

    private fun hzToMel(hz: Double) = 2595 * log10(1 + hz / 700)
    private fun melToHz(mel: Double) = 700 * (10.0.pow(mel / 2595) - 1)
    private fun Double.roundToInt() = kotlin.math.roundToInt(this)
}
```

## 7. WhisperTranscriber.kt — Core Inference Engine

```kotlin
// WhisperTranscriber.kt
package com.yourname.voicenotes

import android.content.Context
import android.os.SystemClock
import android.util.Log
import org.tensorflow.lite.Interpreter
import org.tensorflow.lite.support.common.FileUtil
import java.nio.ByteBuffer
import java.nio.ByteOrder

class WhisperTranscriber(private val context: Context) {

    companion object {
        private const val TAG           = "WhisperTranscriber"
        private const val ENCODER_MODEL = "whisper_tiny_encoder.tflite"
        private const val DECODER_MODEL = "whisper_tiny_decoder.tflite"
        private const val N_MELS        = 80
        private const val N_FRAMES      = 3000
        private const val MAX_TOKENS    = 224    // max output tokens
        private const val SOT_TOKEN     = 50258  // start-of-transcript
        private const val EOT_TOKEN     = 50257  // end-of-transcript
        private const val LANG_EN       = 50259  // English language token
        private const val TRANSCRIBE    = 50359  // transcribe task token
    }

    private val encoder: Interpreter
    private val decoder: Interpreter
    private val vocab: Map<Int, String>

    init {
        val options = Interpreter.Options().apply { setNumThreads(4) }
        encoder = Interpreter(FileUtil.loadMappedFile(context, ENCODER_MODEL), options)
        decoder = Interpreter(FileUtil.loadMappedFile(context, DECODER_MODEL), options)
        vocab   = loadVocab()
        Log.d(TAG, "Whisper loaded — encoder+decoder ready")
    }

    /**
     * Transcribe a Float32 PCM audio array to text.
     * Audio must be 16kHz mono, max 30 seconds.
     */
    fun transcribe(audioFloat: FloatArray, language: String = "en"): TranscriptionResult {
        val totalStartMs = SystemClock.uptimeMillis()

        // Step 1: Compute mel spectrogram
        val melSpec = MelSpectrogram.compute(audioFloat)

        // Step 2: Encode audio features
        val encoderStart = SystemClock.uptimeMillis()
        val audioFeatures = runEncoder(melSpec)
        val encoderMs = SystemClock.uptimeMillis() - encoderStart
        Log.d(TAG, "Encoder: ${encoderMs}ms")

        // Step 3: Decode tokens autoregressively
        val decoderStart = SystemClock.uptimeMillis()
        val tokens = runDecoder(audioFeatures, language)
        val decoderMs = SystemClock.uptimeMillis() - decoderStart
        Log.d(TAG, "Decoder: ${decoderMs}ms, tokens: ${tokens.size}")

        // Step 4: Convert tokens to text
        val text = detokenize(tokens)
        val totalMs = SystemClock.uptimeMillis() - totalStartMs

        return TranscriptionResult(
            text         = text.trim(),
            inferenceMs  = totalMs,
            encoderMs    = encoderMs,
            decoderMs    = decoderMs,
            tokenCount   = tokens.size
        )
    }

    private fun runEncoder(melSpec: Array<Array<FloatArray>>): ByteBuffer {
        // Input: [1, 80, 3000] float mel spectrogram
        val inputBuf = ByteBuffer
            .allocateDirect(1 * N_MELS * N_FRAMES * 4)
            .order(ByteOrder.nativeOrder())

        for (mel in 0 until N_MELS) {
            for (frame in 0 until N_FRAMES) {
                inputBuf.putFloat(melSpec[0][mel][frame])
            }
        }
        inputBuf.rewind()

        // Output: [1, 1500, 384] float audio features (tiny model)
        val outputBuf = ByteBuffer
            .allocateDirect(1 * 1500 * 384 * 4)
            .order(ByteOrder.nativeOrder())

        encoder.run(inputBuf, outputBuf)
        outputBuf.rewind()
        return outputBuf
    }

    private fun runDecoder(audioFeatures: ByteBuffer, language: String): List<Int> {
        // Initial tokens: [SOT, language_token, TRANSCRIBE]
        val langToken = getLanguageToken(language)
        val tokens    = mutableListOf(SOT_TOKEN, langToken, TRANSCRIBE)

        for (step in 0 until MAX_TOKENS) {
            // Input: current token sequence + audio features
            val tokenInput = ByteBuffer
                .allocateDirect(tokens.size * 4)
                .order(ByteOrder.nativeOrder())
            tokens.forEach { tokenInput.putInt(it) }
            tokenInput.rewind()
            audioFeatures.rewind()

            // Output: logits over vocabulary [1, vocab_size ~51865]
            val logitsBuf = ByteBuffer
                .allocateDirect(1 * 51865 * 4)
                .order(ByteOrder.nativeOrder())

            decoder.runForMultipleInputsOutputs(
                arrayOf(tokenInput, audioFeatures),
                mapOf(0 to logitsBuf)
            )

            // Greedy decode: pick highest logit token
            logitsBuf.rewind()
            val logits = FloatArray(51865)
            logitsBuf.asFloatBuffer().get(logits)
            val nextToken = logits.indices.maxByOrNull { logits[it] } ?: break

            if (nextToken == EOT_TOKEN) break
            tokens.add(nextToken)
        }

        return tokens.drop(3)  // remove SOT + lang + TRANSCRIBE prefix
    }

    private fun detokenize(tokens: List<Int>): String {
        return tokens.mapNotNull { vocab[it] }.joinToString("")
            .replace("▁", " ")     // SentencePiece space token
            .replace("<|.*?|>".toRegex(), "")  // remove special tokens
            .trim()
    }

    private fun getLanguageToken(language: String): Int {
        // Whisper language tokens start at 50259
        val languages = mapOf(
            "en" to 50259, "zh" to 50260, "de" to 50261, "es" to 50262,
            "ru" to 50263, "ko" to 50264, "fr" to 50265, "ja" to 50266,
            "pt" to 50267, "tr" to 50268, "ar" to 50269, "hi" to 50277
        )
        return languages[language] ?: LANG_EN
    }

    private fun loadVocab(): Map<Int, String> {
        val vocab = mutableMapOf<Int, String>()
        try {
            context.assets.open("whisper_multilingual_vocab.txt")
                .bufferedReader()
                .forEachLine { line ->
                    val parts = line.split("\t")
                    if (parts.size == 2) {
                        vocab[parts[0].toIntOrNull() ?: return@forEachLine] = parts[1]
                    }
                }
        } catch (e: Exception) {
            Log.w(TAG, "Vocab load failed: ${e.message}")
        }
        return vocab
    }

    fun close() {
        encoder.close()
        decoder.close()
    }
}

data class TranscriptionResult(
    val text: String,
    val inferenceMs: Long,
    val encoderMs: Long,
    val decoderMs: Long,
    val tokenCount: Int
)
```

## 8. Usage in Activity

```kotlin
// TranscriptionActivity.kt (simplified)
class TranscriptionActivity : AppCompatActivity() {

    private lateinit var transcriber: WhisperTranscriber
    private lateinit var recorder: AudioRecorder
    private lateinit var btnRecord: Button
    private lateinit var tvTranscript: TextView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_transcription)

        transcriber = WhisperTranscriber(this)
        recorder    = AudioRecorder(this)
        btnRecord   = findViewById(R.id.btnRecord)
        tvTranscript = findViewById(R.id.tvTranscript)

        btnRecord.setOnClickListener {
            if (recorder.isCurrentlyRecording) {
                recorder.stopRecording()
                btnRecord.text = "🎙️ Record"
            } else {
                startRecordingAndTranscribe()
            }
        }
    }

    @SuppressLint("MissingPermission")
    private fun startRecordingAndTranscribe() {
        btnRecord.text = "⏹ Stop"
        tvTranscript.text = "Recording..."

        lifecycleScope.launch {
            // Record up to 30 seconds (stops when user taps Stop)
            val audio = recorder.record(durationSeconds = 30)

            tvTranscript.text = "Transcribing..."
            val result = withContext(Dispatchers.Default) {
                transcriber.transcribe(audio, language = "en")
            }

            tvTranscript.text = result.text
            Log.d("Whisper", "Done in ${result.inferenceMs}ms: '${result.text}'")
            btnRecord.text = "🎙️ Record"
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        transcriber.close()
    }
}
```

## 10. Summary

- Whisper expects Float32 PCM at 16kHz → 80-channel mel spectrogram → encoder → decoder
- `AudioRecord` captures Int16 PCM — always convert to Float32 before Whisper
- Two-model architecture: encoder (audio → features) + decoder (features → tokens)
- Greedy decoding is the simplest approach — beam search gives better accuracy
- Language is specified via a token (not a separate model config)
