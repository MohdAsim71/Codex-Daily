# 4.7 — 🛠️ Project: Offline Voice Notes App with Transcription

> **Module 4 · Project**  
> 🛠️ Build · ⏱️ ~90 min

---

## 🎯 Project Goal

Build a **production-quality offline voice notes Android app** that:

- Records audio from the microphone
- Transcribes speech to text using Whisper TFLite — fully offline
- Auto-detects sentiment and tags each note (😄 😐 😞)
- Translates notes to any of 50+ languages via ML Kit
- Classifies note topic automatically (work / personal / ideas / health)
- Saves notes locally with Room database
- Works completely without internet — zero cloud dependency

## 4. RecordViewModel.kt — The AI Pipeline

This is the heart of the app — orchestrates recording → transcription → sentiment → topic:

```kotlin
// ui/viewmodel/RecordViewModel.kt
package com.yourname.voicenotes.ui.viewmodel

import android.annotation.SuppressLint
import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.yourname.voicenotes.data.*
import com.yourname.voicenotes.ml.*
import kotlinx.coroutines.*
import kotlinx.coroutines.flow.*
import java.io.File

// Pipeline states shown to the user
sealed class RecordState {
    object Idle        : RecordState()
    object Recording   : RecordState()
    data class Processing(val step: String, val progress: Int) : RecordState()
    data class Done(val note: VoiceNote)                       : RecordState()
    data class Error(val message: String)                      : RecordState()
}

class RecordViewModel(application: Application) : AndroidViewModel(application) {

    private val db         = VoiceNoteDatabase.getInstance(application)
    private val repository = VoiceNoteRepository(db.voiceNoteDao())

    private val recorder    = AudioRecorder(application)
    private val transcriber = WhisperTranscriber(application)
    private val sentiment   = SentimentAnalyzer(application)

    private val _state = MutableStateFlow<RecordState>(RecordState.Idle)
    val state: StateFlow<RecordState> = _state.asStateFlow()

    private val _micLevel = MutableStateFlow(0f)
    val micLevel: StateFlow<Float> = _micLevel.asStateFlow()

    private var recordingJob: Job? = null
    private var recordedAudio: FloatArray? = null

    // ── Recording ──────────────────────────────────────

    @SuppressLint("MissingPermission")
    fun startRecording() {
        recordingJob = viewModelScope.launch(Dispatchers.IO) {
            _state.value = RecordState.Recording
            recordedAudio = recorder.record(durationSeconds = 30)
            // If we get here, recording stopped naturally (30s limit)
            if (_state.value == RecordState.Recording) {
                processRecording()
            }
        }
    }

    fun stopRecording() {
        recorder.stopRecording()
        viewModelScope.launch {
            delay(200)   // brief pause to let recorder finish
            processRecording()
        }
    }

    // ── AI Processing Pipeline ─────────────────────────

    private fun processRecording() {
        val audio = recordedAudio ?: run {
            _state.value = RecordState.Error("No audio recorded")
            return
        }

        viewModelScope.launch(Dispatchers.Default) {

            try {
                // Step 1: Transcribe
                _state.value = RecordState.Processing("Transcribing with Whisper...", 25)
                val transcription = transcriber.transcribe(audio)

                if (transcription.text.isBlank()) {
                    _state.value = RecordState.Error("No speech detected. Try again.")
                    return@launch
                }

                // Step 2: Sentiment analysis
                _state.value = RecordState.Processing("Analyzing sentiment...", 50)
                val sentimentResult = sentiment.analyze(transcription.text)

                // Step 3: Topic classification
                _state.value = RecordState.Processing("Classifying topic...", 75)
                val topic = classifyTopic(transcription.text)

                // Step 4: Save audio file
                _state.value = RecordState.Processing("Saving note...", 90)
                val audioFile = saveAudioFile(audio)

                // Step 5: Save to database
                val note = VoiceNote(
                    audioFilePath    = audioFile,
                    transcript       = transcription.text,
                    sentiment        = mapSentiment(sentimentResult.sentiment),
                    sentimentScore   = sentimentResult.positiveScore,
                    topic            = topic,
                    durationSeconds  = (audio.size / AudioRecorder.SAMPLE_RATE),
                    transcriptionMs  = transcription.inferenceMs
                )

                val savedId = repository.insert(note)
                val savedNote = note.copy(id = savedId)

                _state.value = RecordState.Done(savedNote)

            } catch (e: Exception) {
                _state.value = RecordState.Error("Processing failed: ${e.message}")
            }
        }
    }

    private fun classifyTopic(text: String): NoteTopic {
        val lower = text.lowercase()
        return when {
            containsAny(lower, "meeting", "deadline", "project", "client",
                "email", "boss", "work", "office", "presentation") -> NoteTopic.WORK
            containsAny(lower, "doctor", "health", "exercise", "gym",
                "medicine", "sick", "diet", "sleep", "tired")       -> NoteTopic.HEALTH
            containsAny(lower, "buy", "shop", "store", "price",
                "order", "amazon", "grocery", "market")              -> NoteTopic.SHOPPING
            containsAny(lower, "idea", "think", "what if", "concept",
                "imagine", "could", "maybe", "plan", "vision")       -> NoteTopic.IDEAS
            containsAny(lower, "feel", "family", "friend", "personal",
                "home", "life", "today", "myself")                   -> NoteTopic.PERSONAL
            else -> NoteTopic.GENERAL
        }
    }

    private fun containsAny(text: String, vararg keywords: String) =
        keywords.any { text.contains(it) }

    private fun mapSentiment(s: com.yourname.voicenotes.ml.Sentiment) = when (s) {
        com.yourname.voicenotes.ml.Sentiment.VERY_POSITIVE -> NoteSentiment.VERY_POSITIVE
        com.yourname.voicenotes.ml.Sentiment.POSITIVE      -> NoteSentiment.POSITIVE
        com.yourname.voicenotes.ml.Sentiment.NEUTRAL       -> NoteSentiment.NEUTRAL
        com.yourname.voicenotes.ml.Sentiment.NEGATIVE      -> NoteSentiment.NEGATIVE
        com.yourname.voicenotes.ml.Sentiment.VERY_NEGATIVE -> NoteSentiment.VERY_NEGATIVE
    }

    private fun saveAudioFile(audio: FloatArray): String {
        val file = File(
            getApplication<Application>().filesDir,
            "note_${System.currentTimeMillis()}.pcm"
        )
        file.outputStream().use { out ->
            val bytes = ByteArray(audio.size * 4)
            audio.forEachIndexed { i, f ->
                val bits = java.lang.Float.floatToIntBits(f)
                bytes[i * 4]     = (bits shr 24).toByte()
                bytes[i * 4 + 1] = (bits shr 16).toByte()
                bytes[i * 4 + 2] = (bits shr 8).toByte()
                bytes[i * 4 + 3] = bits.toByte()
            }
            out.write(bytes)
        }
        return file.absolutePath
    }

    fun reset() {
        recordedAudio = null
        _state.value = RecordState.Idle
    }

    override fun onCleared() {
        super.onCleared()
        transcriber.close()
        sentiment.close()
    }
}
```

```kotlin
// ui/adapter/VoiceNoteAdapter.kt
package com.yourname.voicenotes.ui.adapter

import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.TextView
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.google.android.material.button.MaterialButton
import com.yourname.voicenotes.R
import com.yourname.voicenotes.data.VoiceNote
import com.yourname.voicenotes.utils.TimeUtils

class VoiceNoteAdapter(
    private val onTranslateClick: (VoiceNote) -> Unit,
    private val onDeleteClick:    (VoiceNote) -> Unit
) : ListAdapter<VoiceNote, VoiceNoteAdapter.ViewHolder>(DIFF_CALLBACK) {

    inner class ViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val tvSentimentEmoji: TextView = view.findViewById(R.id.tvSentimentEmoji)
        val tvTopic:          TextView = view.findViewById(R.id.tvTopic)
        val tvTime:           TextView = view.findViewById(R.id.tvTime)
        val tvTranscript:     TextView = view.findViewById(R.id.tvTranscript)
        val btnTranslate: MaterialButton = view.findViewById(R.id.btnTranslate)
        val btnDelete:    MaterialButton = view.findViewById(R.id.btnDelete)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_voice_note, parent, false)
        return ViewHolder(view)
    }

    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val note = getItem(position)

        holder.tvSentimentEmoji.text = note.sentiment.emoji
        holder.tvTopic.text          = "${note.topic.emoji} ${note.topic.label}  ·  ${note.durationSeconds}s"
        holder.tvTime.text           = TimeUtils.relativeTime(note.createdAt)
        holder.tvTranscript.text     = note.transcript

        holder.btnTranslate.setOnClickListener { onTranslateClick(note) }
        holder.btnDelete.setOnClickListener    { onDeleteClick(note) }
    }

    companion object {
        private val DIFF_CALLBACK = object : DiffUtil.ItemCallback<VoiceNote>() {
            override fun areItemsTheSame(old: VoiceNote, new: VoiceNote) = old.id == new.id
            override fun areContentsTheSame(old: VoiceNote, new: VoiceNote) = old == new
        }
    }
}
```

```kotlin
package com.yourname.voicenotes.ui

import android.os.Bundle
import android.view.View

import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import com.google.android.material.button.MaterialButton
import com.google.android.material.progressindicator.LinearProgressIndicator
import com.yourname.voicenotes.R
import com.yourname.voicenotes.ui.viewmodel.RecordState
import com.yourname.voicenotes.ui.viewmodel.RecordViewModel
import kotlinx.coroutines.launch

class RecordActivity : AppCompatActivity() {

    private val viewModel: RecordViewModel by viewModels()

    private lateinit var tvStatus:    TextView
    private lateinit var tvSubStatus: TextView
    private lateinit var btnRecord:   MaterialButton
    private lateinit var progressBar: LinearProgressIndicator

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_record)

        tvStatus    = findViewById(R.id.tvStatus)
        tvSubStatus = findViewById(R.id.tvSubStatus)
        btnRecord   = findViewById(R.id.btnRecord)
        progressBar = findViewById(R.id.progressBar)

        observeState()
        setupRecordButton()
    }

    private fun setupRecordButton() {
        btnRecord.setOnClickListener {
            when (viewModel.state.value) {
                is RecordState.Idle -> {
                    // Check RECORD_AUDIO permission first in production
                    viewModel.startRecording()
                }
                is RecordState.Recording -> {
                    viewModel.stopRecording()
                }
                is RecordState.Done, is RecordState.Error -> {
                    viewModel.reset()
                }
                else -> {}
            }
        }
    }

    private fun observeState() {
        lifecycleScope.launch {
            viewModel.state.collect { state ->
                when (state) {
                    RecordState.Idle -> {
                        tvStatus.text    = "Ready to record"
                        tvSubStatus.text = "Tap the button to start"
                        btnRecord.text   = "🎙️  Record"
                        progressBar.visibility = View.GONE
                    }

                    RecordState.Recording -> {
                        tvStatus.text    = "🔴 Recording..."
                        tvSubStatus.text = "Tap Stop when done (max 30s)"
                        btnRecord.text   = "⏹  Stop"
                        progressBar.visibility = View.GONE
                    }

                    is RecordState.Processing -> {
                        tvStatus.text    = state.step
                        tvSubStatus.text = "${state.progress}% complete"
                        btnRecord.isEnabled    = false
                        progressBar.visibility = View.VISIBLE
                        progressBar.progress   = state.progress
                    }

                    is RecordState.Done -> {
                        val note = state.note
                        tvStatus.text    = "✅ Note saved!"
                        tvSubStatus.text = buildString {
                            append("\"${note.transcript.take(60)}")
                            if (note.transcript.length > 60) append("...")
                            append("\"\n")
                            append("${note.sentiment.emoji} ${note.sentiment.label}  ·  ")
                            append("${note.topic.emoji} ${note.topic.label}  ·  ")
                            append("${note.transcriptionMs}ms")
                        }
                        btnRecord.text      = "🎙️  Record Another"
                        btnRecord.isEnabled = true
                        progressBar.visibility = View.GONE
                    }

                    is RecordState.Error -> {
                        tvStatus.text    = "❌ ${state.message}"
                        tvSubStatus.text = "Tap to try again"
                        btnRecord.text      = "🎙️  Try Again"
                        btnRecord.isEnabled = true
                        progressBar.visibility = View.GONE
                    }
                }
            }
        }
    }
}
```

```kotlin
// ui/MainActivity.kt
package com.yourname.voicenotes.ui

import android.content.Intent
import android.os.Bundle
import android.widget.Toast
import androidx.activity.viewModels
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.google.android.material.dialog.MaterialAlertDialogBuilder
import com.google.android.material.floatingactionbutton.FloatingActionButton
import com.yourname.voicenotes.R
import com.yourname.voicenotes.data.VoiceNote
import com.yourname.voicenotes.ml.TranslationManager
import com.yourname.voicenotes.ui.adapter.VoiceNoteAdapter
import com.yourname.voicenotes.ui.viewmodel.MainViewModel
import kotlinx.coroutines.launch

class MainActivity : AppCompatActivity() {

    private val viewModel: MainViewModel by viewModels()
    private lateinit var adapter: VoiceNoteAdapter
    private val translationManager = TranslationManager()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        setupRecyclerView()
        observeNotes()

        findViewById<FloatingActionButton>(R.id.fabRecord).setOnClickListener {
            startActivity(Intent(this, RecordActivity::class.java))
        }
    }

    private fun setupRecyclerView() {
        adapter = VoiceNoteAdapter(
            onTranslateClick = { note -> showTranslateDialog(note) },
            onDeleteClick    = { note -> confirmDelete(note) }
        )

        findViewById<RecyclerView>(R.id.recyclerView).apply {
            layoutManager = LinearLayoutManager(this@MainActivity)
            adapter = this@MainActivity.adapter
        }
    }

    private fun observeNotes() {
        lifecycleScope.launch {
            viewModel.allNotes.collect { notes ->
                adapter.submitList(notes)
            }
        }
    }

    private fun showTranslateDialog(note: VoiceNote) {
        val languages = TranslationManager.SUPPORTED_LANGUAGES.entries
            .sortedBy { it.value }
        val displayNames = languages.map { "${it.value} (${it.key})" }.toTypedArray()

        MaterialAlertDialogBuilder(this)
            .setTitle("Translate to...")
            .setItems(displayNames) { _, which ->
                val targetLang = languages[which].key
                translateNote(note, targetLang)
            }
            .show()
    }

    private fun translateNote(note: VoiceNote, targetLang: String) {
        val langName = TranslationManager.SUPPORTED_LANGUAGES[targetLang] ?: targetLang
        Toast.makeText(this, "Translating to $langName...", Toast.LENGTH_SHORT).show()

        lifecycleScope.launch {
            try {
                val translated = translationManager.translate(
                    text       = note.transcript,
                    sourceLang = "en",
                    targetLang = targetLang
                )
                MaterialAlertDialogBuilder(this@MainActivity)
                    .setTitle("Translation to $langName")
                    .setMessage(translated)
                    .setPositiveButton("OK", null)
                    .show()
            } catch (e: Exception) {
                Toast.makeText(this@MainActivity,
                    "Translation failed: ${e.message}", Toast.LENGTH_SHORT).show()
            }
        }
    }

    private fun confirmDelete(note: VoiceNote) {
        MaterialAlertDialogBuilder(this)
            .setTitle("Delete Note?")
            .setMessage("This note will be permanently deleted.")
            .setPositiveButton("Delete") { _, _ -> viewModel.deleteNote(note) }
            .setNegativeButton("Cancel", null)
            .show()
    }

    override fun onDestroy() {
        super.onDestroy()
        translationManager.close()
    }
}
```

```kotlin
// utils/TimeUtils.kt
package com.yourname.voicenotes.utils

object TimeUtils {

    fun relativeTime(timestamp: Long): String {
        val diff = System.currentTimeMillis() - timestamp
        return when {
            diff < 60_000L          -> "just now"
            diff < 3_600_000L       -> "${diff / 60_000} min ago"
            diff < 86_400_000L      -> "${diff / 3_600_000} hr ago"
            diff < 604_800_000L     -> "${diff / 86_400_000} days ago"
            else -> {
                val sdf = java.text.SimpleDateFormat("MMM d", java.util.Locale.getDefault())
                sdf.format(java.util.Date(timestamp))
            }
        }
    }
}
```
