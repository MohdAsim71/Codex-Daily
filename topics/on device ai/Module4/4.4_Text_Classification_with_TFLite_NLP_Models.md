## 2. Get the Model (Python)

Export DistilBERT sentiment classifier to TFLite:

```python
# export_distilbert_tflite.py
# pip install transformers tensorflow

from transformers import TFDistilBertForSequenceClassification, DistilBertTokenizer
import tensorflow as tf

MODEL_NAME = "distilbert-base-uncased-finetuned-sst-2-english"
MAX_LEN    = 128

# Load model and tokenizer
tokenizer = DistilBertTokenizer.from_pretrained(MODEL_NAME)
model     = TFDistilBertForSequenceClassification.from_pretrained(MODEL_NAME)

# Create a concrete function for export
@tf.function(input_signature=[
    tf.TensorSpec(shape=[1, MAX_LEN], dtype=tf.int32, name="input_ids"),
    tf.TensorSpec(shape=[1, MAX_LEN], dtype=tf.int32, name="attention_mask"),
])
def serving_fn(input_ids, attention_mask):
    return model(input_ids=input_ids, attention_mask=attention_mask)

# Convert to TFLite
converter = tf.lite.TFLiteConverter.from_concrete_functions(
    [serving_fn.get_concrete_function()]
)
converter.optimizations = [tf.lite.Optimize.DEFAULT]  # dynamic range quantization
tflite_model = converter.convert()

with open("distilbert_sentiment.tflite", "wb") as f:
    f.write(tflite_model)

# Save vocabulary for Android
vocab = tokenizer.get_vocab()
with open("distilbert_vocab.txt", "w", encoding="utf-8") as f:
    for token, idx in sorted(vocab.items(), key=lambda x: x[1]):
        f.write(f"{token}\n")

print(f"Model size: {len(tflite_model)/1e6:.1f} MB")
print(f"Vocabulary: {len(vocab)} tokens")
```

Copy to assets:
```
app/src/main/assets/
├── distilbert_sentiment.tflite   ← ~67 MB FP32, ~17 MB quantized
└── distilbert_vocab.txt          ← 30,522 tokens
```

## 3. WordPieceTokenizer.kt

BERT uses WordPiece tokenization. We implement it in Kotlin:

```kotlin
// WordPieceTokenizer.kt
package com.yourname.voicenotes

import android.content.Context

class WordPieceTokenizer(context: Context, vocabFile: String = "distilbert_vocab.txt") {

    companion object {
        const val UNK_TOKEN = "[UNK]"
        const val CLS_TOKEN = "[CLS]"
        const val SEP_TOKEN = "[SEP]"
        const val PAD_TOKEN = "[PAD]"
        const val MAX_INPUT_CHARS_PER_WORD = 200
    }

    private val vocab: Map<String, Int>
    private val ids2Token: Map<Int, String>

    val clsId: Int get() = vocab[CLS_TOKEN] ?: 101
    val sepId: Int get() = vocab[SEP_TOKEN] ?: 102
    val padId: Int get() = vocab[PAD_TOKEN] ?: 0
    val unkId: Int get() = vocab[UNK_TOKEN] ?: 100

    init {
        val vocabList = context.assets.open(vocabFile)
            .bufferedReader()
            .readLines()
            .filter { it.isNotBlank() }

        vocab     = vocabList.mapIndexed { idx, token -> token to idx }.toMap()
        ids2Token = vocabList.mapIndexed { idx, token -> idx to token }.toMap()
    }

    /**
     * Tokenize text into [input_ids, attention_mask] ready for BERT inference.
     * Automatically adds [CLS] and [SEP] tokens, pads/truncates to maxLen.
     */
    fun tokenize(text: String, maxLen: Int = 128): TokenizerOutput {
        val words  = basicTokenize(text)
        val tokens = mutableListOf(clsId)     // [CLS]

        for (word in words) {
            val wordTokens = wordPieceTokenize(word)
            if (tokens.size + wordTokens.size + 1 > maxLen) break  // +1 for [SEP]
            tokens.addAll(wordTokens)
        }

        tokens.add(sepId)   // [SEP]

        // Pad to maxLen
        val inputIds      = IntArray(maxLen) { if (it < tokens.size) tokens[it] else padId }
        val attentionMask = IntArray(maxLen) { if (it < tokens.size) 1 else 0 }

        return TokenizerOutput(inputIds, attentionMask)
    }

    /** Basic tokenization: lowercase + punctuation splitting */
    private fun basicTokenize(text: String): List<String> {
        return text.lowercase()
            .replace(Regex("[^a-z0-9\\s'\\-]"), " ")  // keep alphanumeric + apostrophe
            .trim()
            .split("\\s+".toRegex())
            .filter { it.isNotBlank() }
    }

    /** WordPiece sub-word tokenization */
    private fun wordPieceTokenize(word: String): List<Int> {
        if (word.length > MAX_INPUT_CHARS_PER_WORD) return listOf(unkId)

        val subTokenIds = mutableListOf<Int>()
        var start = 0

        while (start < word.length) {
            var end = word.length
            var curSubStr: String? = null

            while (start < end) {
                val subStr = if (start == 0) word.substring(start, end)
                             else "##${word.substring(start, end)}"
                if (vocab.containsKey(subStr)) {
                    curSubStr = subStr
                    break
                }
                end--
            }

            if (curSubStr == null) return listOf(unkId)
            subTokenIds.add(vocab[curSubStr]!!)
            start = end
        }

        return subTokenIds
    }

    fun decode(ids: List<Int>): String {
        return ids.mapNotNull { ids2Token[it] }
            .joinToString(" ")
            .replace(" ##", "")
            .replace(Regex("\\[.*?\\]"), "")
            .trim()
    }
}

data class TokenizerOutput(
    val inputIds: IntArray,
    val attentionMask: IntArray
)
```

## 4. TextClassifier.kt — Full Implementation

```kotlin
// TextClassifier.kt
package com.yourname.voicenotes

import android.content.Context
import android.os.SystemClock
import android.util.Log
import org.tensorflow.lite.Interpreter
import org.tensorflow.lite.support.common.FileUtil
import java.nio.ByteBuffer
import java.nio.ByteOrder

data class ClassificationResult(
    val label: String,
    val confidence: Float,
    val allScores: Map<String, Float>,
    val inferenceMs: Long
)

class TextClassifier(
    context: Context,
    modelFile: String = "distilbert_sentiment.tflite",
    private val labels: List<String> = listOf("NEGATIVE", "POSITIVE"),
    private val maxLen: Int = 128
) {
    companion object { private const val TAG = "TextClassifier" }

    private val interpreter: Interpreter
    private val tokenizer: WordPieceTokenizer

    init {
        val options = Interpreter.Options().setNumThreads(4)
        interpreter = Interpreter(FileUtil.loadMappedFile(context, modelFile), options)
        tokenizer   = WordPieceTokenizer(context)

        Log.d(TAG, "TextClassifier ready")
        Log.d(TAG, "  Input[0] shape : ${interpreter.getInputTensor(0).shape().toList()}")
        Log.d(TAG, "  Input[1] shape : ${interpreter.getInputTensor(1).shape().toList()}")
        Log.d(TAG, "  Output shape   : ${interpreter.getOutputTensor(0).shape().toList()}")
    }

    fun classify(text: String): ClassificationResult {
        // Step 1: Tokenize
        val tokenized = tokenizer.tokenize(text, maxLen)

        // Step 2: Build input ByteBuffers
        val inputIdsBuf = intArrayToByteBuffer(tokenized.inputIds)
        val maskBuf     = intArrayToByteBuffer(tokenized.attentionMask)

        // Step 3: Prepare output buffer [1, numClasses]
        val outputBuf = ByteBuffer
            .allocateDirect(1 * labels.size * 4)
            .order(ByteOrder.nativeOrder())

        // Step 4: Run inference (two inputs for BERT-style models)
        val startMs = SystemClock.uptimeMillis()
        interpreter.runForMultipleInputsOutputs(
            arrayOf(inputIdsBuf, maskBuf),
            mapOf(0 to outputBuf)
        )
        val inferenceMs = SystemClock.uptimeMillis() - startMs

        // Step 5: Decode logits → probabilities
        outputBuf.rewind()
        val logits = FloatArray(labels.size)
        outputBuf.asFloatBuffer().get(logits)
        val probs = softmax(logits)

        val bestIdx   = probs.indices.maxByOrNull { probs[it] }!!
        val allScores = labels.zip(probs.toList()).toMap()

        return ClassificationResult(
            label       = labels[bestIdx],
            confidence  = probs[bestIdx],
            allScores   = allScores,
            inferenceMs = inferenceMs
        )
    }

    private fun intArrayToByteBuffer(array: IntArray): ByteBuffer {
        val buf = ByteBuffer
            .allocateDirect(array.size * 4)
            .order(ByteOrder.nativeOrder())
        array.forEach { buf.putInt(it) }
        buf.rewind()
        return buf
    }

    private fun softmax(logits: FloatArray): FloatArray {
        val maxLogit = logits.max()
        val exps = logits.map { Math.exp((it - maxLogit).toDouble()).toFloat() }
        val sum  = exps.sum()
        return exps.map { it / sum }.toFloatArray()
    }

    fun close() = interpreter.close()
}
```

## 5. Usage Example

```kotlin
// In your ViewModel or Activity
val classifier = TextClassifier(context)

// Classify user input or transcribed speech
val result = withContext(Dispatchers.Default) {
    classifier.classify("Running AI locally is surprisingly fast and private!")
}

// result.label      → "POSITIVE"
// result.confidence → 0.987
// result.inferenceMs → 43
// result.allScores  → {NEGATIVE=0.013, POSITIVE=0.987}
```

## 7. Performance Reference

| Model | Quantization | Size | Latency (CPU) |
|---|---|---|---|
| DistilBERT FP32 | None | 255 MB | ~200ms |
| DistilBERT Dynamic INT8 | Dynamic range | ~65 MB | ~90ms |
| DistilBERT Full INT8 | PTQ INT8 | ~33 MB | ~50ms |
| TinyBERT | Distilled | ~55 MB | ~40ms |
| MobileBERT | Mobile optimized | ~95 MB | ~60ms |

> 💡 For production, use full INT8 quantization. 50ms is imperceptible to users and the size savings are significant.

---

## 8. Summary

- BERT-style models need two inputs: `input_ids` and `attention_mask`
- WordPiece tokenization: `[CLS]` + sub-word tokens + `[SEP]` + padding
- Use `runForMultipleInputsOutputs()` for multi-input TFLite models
- Always apply softmax — BERT outputs raw logits, not probabilities
- For custom categories: fine-tune in Python, change `labels` list in Kotlin
