## 3. TranslationManager.kt — Core Implementation

```kotlin
// TranslationManager.kt
package com.yourname.voicenotes

import android.util.Log
import com.google.mlkit.common.model.DownloadConditions
import com.google.mlkit.common.model.RemoteModelManager
import com.google.mlkit.nl.languageid.LanguageIdentification
import com.google.mlkit.nl.translate.*
import kotlinx.coroutines.suspendCancellableCoroutine
import kotlin.coroutines.resume
import kotlin.coroutines.resumeWithException

class TranslationManager {

    companion object {
        private const val TAG = "TranslationManager"

        // All 50+ supported language codes
        val SUPPORTED_LANGUAGES = mapOf(
            "af" to "Afrikaans",   "ar" to "Arabic",       "be" to "Belarusian",
            "bg" to "Bulgarian",   "bn" to "Bengali",      "ca" to "Catalan",
            "cs" to "Czech",       "cy" to "Welsh",        "da" to "Danish",
            "de" to "German",      "el" to "Greek",        "en" to "English",
            "eo" to "Esperanto",   "es" to "Spanish",      "et" to "Estonian",
            "fa" to "Persian",     "fi" to "Finnish",      "fr" to "French",
            "ga" to "Irish",       "gl" to "Galician",     "gu" to "Gujarati",
            "he" to "Hebrew",      "hi" to "Hindi",        "hr" to "Croatian",
            "ht" to "Haitian Creole","hu" to "Hungarian",  "id" to "Indonesian",
            "is" to "Icelandic",   "it" to "Italian",      "ja" to "Japanese",
            "ka" to "Georgian",    "kn" to "Kannada",      "ko" to "Korean",
            "lt" to "Lithuanian",  "lv" to "Latvian",      "mk" to "Macedonian",
            "mr" to "Marathi",     "ms" to "Malay",        "mt" to "Maltese",
            "nl" to "Dutch",       "no" to "Norwegian",    "pl" to "Polish",
            "pt" to "Portuguese",  "ro" to "Romanian",     "ru" to "Russian",
            "sk" to "Slovak",      "sl" to "Slovenian",    "sq" to "Albanian",
            "sr" to "Serbian",     "sv" to "Swedish",      "sw" to "Swahili",
            "ta" to "Tamil",       "te" to "Telugu",       "th" to "Thai",
            "tl" to "Filipino",    "tr" to "Turkish",      "uk" to "Ukrainian",
            "ur" to "Urdu",        "vi" to "Vietnamese",   "zh" to "Chinese",
        )
    }

    // Cache translator instances to avoid recreation cost (~200ms)
    private val translatorCache = mutableMapOf<String, Translator>()
    private val languageIdentifier = LanguageIdentification.getClient()

    /**
     * Translate text from [sourceLang] to [targetLang].
     * Downloads language model on first use (~15 MB).
     * Both params are BCP-47 codes: "en", "fr", "hi", "zh", etc.
     */
    suspend fun translate(
        text: String,
        sourceLang: String,
        targetLang: String
    ): String = suspendCancellableCoroutine { continuation ->

        val cacheKey   = "${sourceLang}_${targetLang}"
        val translator = translatorCache.getOrPut(cacheKey) {
            val options = TranslatorOptions.Builder()
                .setSourceLanguage(TranslateLanguage.fromLanguageTag(sourceLang)
                    ?: TranslateLanguage.ENGLISH)
                .setTargetLanguage(TranslateLanguage.fromLanguageTag(targetLang)
                    ?: TranslateLanguage.ENGLISH)
                .build()
            Translation.getClient(options)
        }

        // Ensure model is downloaded (no-op if already present)
        val conditions = DownloadConditions.Builder()
            .requireWifi()   // only download over WiFi
            .build()

        translator.downloadModelIfNeeded(conditions)
            .addOnSuccessListener {
                // Model ready — run translation
                translator.translate(text)
                    .addOnSuccessListener  { translated -> continuation.resume(translated) }
                    .addOnFailureListener  { e -> continuation.resumeWithException(e) }
            }
            .addOnFailureListener { e ->
                Log.e(TAG, "Model download failed: ${e.message}")
                continuation.resumeWithException(e)
            }
    }

    /**
     * Auto-detect the language of [text].
     * Returns BCP-47 language code (e.g. "en", "fr") or "und" if undetermined.
     */
    suspend fun detectLanguage(text: String): String =
        suspendCancellableCoroutine { continuation ->
            languageIdentifier.identifyLanguage(text)
                .addOnSuccessListener  { lang -> continuation.resume(lang) }
                .addOnFailureListener  { e    -> continuation.resume("und") }
        }

    /**
     * Detect language then translate — full auto-translate pipeline.
     */
    suspend fun autoTranslate(text: String, targetLang: String): Pair<String, String> {
        val detectedLang = detectLanguage(text)
        val sourceLang   = if (detectedLang == "und") "en" else detectedLang
        val translated   = translate(text, sourceLang, targetLang)
        return sourceLang to translated
    }

    /**
     * Get list of downloaded language models on this device.
     */
    suspend fun getDownloadedModels(): List<String> =
        suspendCancellableCoroutine { continuation ->
            RemoteModelManager.getInstance()
                .getDownloadedModels(TranslateRemoteModel::class.java)
                .addOnSuccessListener { models ->
                    val codes = models.map { it.language }
                    continuation.resume(codes)
                }
                .addOnFailureListener { continuation.resume(emptyList()) }
        }

    /**
     * Delete a downloaded language model to free storage.
     */
    fun deleteModel(languageCode: String) {
        val model = TranslateRemoteModel.Builder(languageCode).build()
        RemoteModelManager.getInstance().deleteDownloadedModel(model)
            .addOnSuccessListener { Log.d(TAG, "Deleted model: $languageCode") }
            .addOnFailureListener { Log.e(TAG, "Delete failed: ${it.message}") }
    }

    fun close() {
        translatorCache.values.forEach { it.close() }
        translatorCache.clear()
        languageIdentifier.close()
    }
}
```

## 4. ViewModel Integration

```kotlin
// In TranslationViewModel.kt
class TranslationViewModel : ViewModel() {

    private val translationManager = TranslationManager()

    private val _translationState = MutableStateFlow<TranslationState>(TranslationState.Idle)
    val translationState = _translationState.asStateFlow()

    var sourceLang = "en"
    var targetLang = "hi"

    fun translateText(input: String) {
        if (input.isBlank()) return
        _translationState.value = TranslationState.Translating

        viewModelScope.launch {
            try {
                val result = translationManager.translate(input, sourceLang, targetLang)
                _translationState.value = TranslationState.Success(result)
            } catch (e: Exception) {
                _translationState.value = TranslationState.Error(e.message ?: "Translation failed")
            }
        }
    }

    fun autoTranslate(input: String) {
        viewModelScope.launch {
            _translationState.value = TranslationState.Translating
            try {
                val (detected, result) = translationManager.autoTranslate(input, targetLang)
                sourceLang = detected
                _translationState.value = TranslationState.Success(result, detectedLang = detected)
            } catch (e: Exception) {
                _translationState.value = TranslationState.Error(e.message ?: "Error")
            }
        }
    }

    override fun onCleared() {
        super.onCleared()
        translationManager.close()
    }
}

sealed class TranslationState {
    object Idle : TranslationState()
    object Translating : TranslationState()
    data class Success(val text: String, val detectedLang: String = "") : TranslationState()
    data class Error(val message: String) : TranslationState()
}
```

## 5. Language Selector UI

```kotlin
// Language picker — Spinner with all supported languages
fun setupLanguageSpinner(spinner: Spinner, context: Context, onSelected: (String) -> Unit) {
    val entries = TranslationManager.SUPPORTED_LANGUAGES.entries.sortedBy { it.value }
    val displayNames = entries.map { "${it.value} (${it.key})" }
    val codes = entries.map { it.key }

    val adapter = ArrayAdapter(context, android.R.layout.simple_spinner_item, displayNames)
    adapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)
    spinner.adapter = adapter

    spinner.onItemSelectedListener = object : AdapterView.OnItemSelectedListener {
        override fun onItemSelected(parent: AdapterView<*>?, view: View?, position: Int, id: Long) {
            onSelected(codes[position])
        }
        override fun onNothingSelected(parent: AdapterView<*>?) {}
    }
}
```

## 6. Model Download Progress UI

Show download status for the first-time model download:

```kotlin
// Show progress during first-time model download
fun showDownloadProgress(context: Context, targetLang: String) {
    val langName = TranslationManager.SUPPORTED_LANGUAGES[targetLang] ?: targetLang

    // Check if model is already downloaded
    val model = TranslateRemoteModel.Builder(targetLang).build()
    RemoteModelManager.getInstance().isModelDownloaded(model)
        .addOnSuccessListener { isDownloaded ->
            if (!isDownloaded) {
                // Show a Snackbar or dialog
                Snackbar.make(
                    rootView,
                    "Downloading $langName model (~15 MB)...",
                    Snackbar.LENGTH_INDEFINITE
                ).show()
            }
        }
}
```

## 7. Supported Language Pairs

ML Kit supports **all 50×50 = 2,500 language pairs** (via English as pivot):

```
Source: any of 50 languages
Target: any of 50 languages
Total: 2,500 translation pairs

Model download per direction: ~15 MB
Example: English→Hindi = 15 MB download
         Hindi→German  = 30 MB (Hindi model + German model)

Already on device = 0 MB = instant translation
```

## 8. Performance Reference

| Scenario | Latency |
|---|---|
| Model already downloaded, short text | < 50ms |
| Model already downloaded, long paragraph | 50–200ms |
| First use — model download (WiFi) | 5–30 seconds |
| Language detection (any text) | < 20ms |

---

## 9. Summary

- ML Kit Translate is the simplest path to offline translation — no model files to manage
- Language models download once (~15 MB each) then work forever offline
- Always cache `Translator` instances — creating one takes ~200ms
- `LanguageIdentification` detects source language in <20ms — always use this for auto-translate
- Pivot through English for rare language pairs (e.g. Swahili→Gujarati goes via English)
- Respect users' data by using `requireWifi()` for the initial model download
