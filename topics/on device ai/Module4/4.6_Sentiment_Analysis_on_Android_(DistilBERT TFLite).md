## 3. SentimentAnalyzer.kt — Full Implementation

Wraps `TextClassifier` with sentiment-specific logic:

```kotlin
// SentimentAnalyzer.kt
package com.yourname.voicenotes

import android.content.Context
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext

enum class Sentiment(val emoji: String, val label: String) {
    VERY_POSITIVE("😄", "Very Positive"),
    POSITIVE      ("🙂", "Positive"),
    NEUTRAL        ("😐", "Neutral"),
    NEGATIVE       ("😕", "Negative"),
    VERY_NEGATIVE  ("😞", "Very Negative")
}

data class SentimentResult(
    val sentiment: Sentiment,
    val confidence: Float,
    val positiveScore: Float,
    val negativeScore: Float,
    val inferenceMs: Long,
    val textSnippet: String
)

class SentimentAnalyzer(context: Context) {

    // Reuse TextClassifier from Lesson 4.4
    private val classifier = TextClassifier(
        context    = context,
        modelFile  = "distilbert_sentiment.tflite",
        labels     = listOf("NEGATIVE", "POSITIVE"),
        maxLen     = 128
    )

    /**
     * Analyze sentiment of [text].
     * Returns a [SentimentResult] with mapped Sentiment enum.
     */
    suspend fun analyze(text: String): SentimentResult = withContext(Dispatchers.Default) {
        val result = classifier.classify(text)

        val positiveScore = result.allScores["POSITIVE"] ?: 0f
        val negativeScore = result.allScores["NEGATIVE"] ?: 0f

        // Map binary positive/negative to 5-class sentiment
        val sentiment = when {
            positiveScore >= 0.90f -> Sentiment.VERY_POSITIVE
            positiveScore >= 0.65f -> Sentiment.POSITIVE
            positiveScore >= 0.40f -> Sentiment.NEUTRAL
            positiveScore >= 0.15f -> Sentiment.NEGATIVE
            else                   -> Sentiment.VERY_NEGATIVE
        }

        SentimentResult(
            sentiment      = sentiment,
            confidence     = if (positiveScore >= 0.5f) positiveScore else negativeScore,
            positiveScore  = positiveScore,
            negativeScore  = negativeScore,
            inferenceMs    = result.inferenceMs,
            textSnippet    = text.take(80)
        )
    }

    /**
     * Lightweight rule-based sentiment — instant, no model needed.
     * Use as a fast preview before running DistilBERT.
     */
    fun quickSentiment(text: String): Sentiment {
        val lower = text.lowercase()
        val positiveWords = setOf(
            "great", "amazing", "wonderful", "love", "happy", "excellent",
            "fantastic", "good", "best", "awesome", "perfect", "brilliant",
            "outstanding", "superb", "joy", "glad", "pleased", "delightful"
        )
        val negativeWords = setOf(
            "bad", "terrible", "awful", "hate", "sad", "horrible", "worst",
            "poor", "disappointing", "failed", "broken", "angry", "upset",
            "disgusting", "frustrating", "annoying", "dreadful", "miserable"
        )

        val words    = lower.split("\\W+".toRegex())
        val posCount = words.count { it in positiveWords }
        val negCount = words.count { it in negativeWords }

        return when {
            posCount > negCount + 1 -> Sentiment.POSITIVE
            negCount > posCount + 1 -> Sentiment.NEGATIVE
            posCount > 0            -> Sentiment.POSITIVE
            negCount > 0            -> Sentiment.NEGATIVE
            else                    -> Sentiment.NEUTRAL
        }
    }

    fun close() = classifier.close()
}
```

## 4. Real-Time Sentiment as User Types

Show sentiment updating live in the UI using a debounced TextWatcher:

```kotlin
// SentimentViewModel.kt
class SentimentViewModel(application: Application) : AndroidViewModel(application) {

    private val analyzer = SentimentAnalyzer(application)

    private val _sentimentResult = MutableStateFlow<SentimentResult?>(null)
    val sentimentResult = _sentimentResult.asStateFlow()

    private val _isAnalyzing = MutableStateFlow(false)
    val isAnalyzing = _isAnalyzing.asStateFlow()

    // Debounce — wait 500ms after user stops typing before running inference
    private var analysisJob: kotlinx.coroutines.Job? = null

    fun onTextChanged(text: String) {
        if (text.isBlank()) {
            _sentimentResult.value = null
            return
        }

        // Show quick rule-based result instantly
        val quick = analyzer.quickSentiment(text)

        // Cancel previous job (debounce)
        analysisJob?.cancel()
        analysisJob = viewModelScope.launch {
            kotlinx.coroutines.delay(500)   // wait 500ms for user to stop typing
            _isAnalyzing.value = true
            try {
                _sentimentResult.value = analyzer.analyze(text)
            } finally {
                _isAnalyzing.value = false
            }
        }
    }

    override fun onCleared() {
        super.onCleared()
        analyzer.close()
    }
}
```

## 6. SentimentActivity.kt

```kotlin
// SentimentActivity.kt
class SentimentActivity : AppCompatActivity() {

    private val viewModel: SentimentViewModel by viewModels()

    private lateinit var etInput: TextInputEditText
    private lateinit var sentimentCard: CardView
    private lateinit var tvEmoji: TextView
    private lateinit var tvSentimentLabel: TextView
    private lateinit var tvConfidence: TextView
    private lateinit var pbPositive: ProgressBar
    private lateinit var pbNegative: ProgressBar
    private lateinit var tvStats: TextView
    private lateinit var progressBar: ProgressBar

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_sentiment)

        bindViews()
        observeState()

        etInput.addTextChangedListener(object : TextWatcher {
            override fun afterTextChanged(s: Editable?) {
                viewModel.onTextChanged(s?.toString() ?: "")
            }
            override fun beforeTextChanged(s: CharSequence?, start: Int, count: Int, after: Int) {}
            override fun onTextChanged(s: CharSequence?, start: Int, before: Int, count: Int) {}
        })
    }

    private fun observeState() {
        lifecycleScope.launch {
            viewModel.sentimentResult.collect { result ->
                if (result != null) {
                    showResult(result)
                } else {
                    sentimentCard.visibility = View.GONE
                }
            }
        }

        lifecycleScope.launch {
            viewModel.isAnalyzing.collect { analyzing ->
                progressBar.visibility = if (analyzing) View.VISIBLE else View.GONE
            }
        }
    }

    private fun showResult(result: SentimentResult) {
        sentimentCard.visibility = View.VISIBLE

        tvEmoji.text          = result.sentiment.emoji
        tvSentimentLabel.text = result.sentiment.label
        tvConfidence.text     = "${(result.confidence * 100).toInt()}% confidence"

        pbPositive.progress = (result.positiveScore * 100).toInt()
        pbNegative.progress = (result.negativeScore * 100).toInt()

        tvStats.text = "⏱ ${result.inferenceMs}ms  ·  DistilBERT  ·  on-device"

        // Color the card based on sentiment
        val cardColor = when (result.sentiment) {
            Sentiment.VERY_POSITIVE -> "#E8F5E9"
            Sentiment.POSITIVE      -> "#F1F8E9"
            Sentiment.NEUTRAL       -> "#FAFAFA"
            Sentiment.NEGATIVE      -> "#FFF3E0"
            Sentiment.VERY_NEGATIVE -> "#FFEBEE"
        }
        sentimentCard.setCardBackgroundColor(Color.parseColor(cardColor))
    }

    private fun bindViews() {
        etInput          = findViewById(R.id.etInput)
        sentimentCard    = findViewById(R.id.sentimentCard)
        tvEmoji          = findViewById(R.id.tvEmoji)
        tvSentimentLabel = findViewById(R.id.tvSentimentLabel)
        tvConfidence     = findViewById(R.id.tvConfidence)
        pbPositive       = findViewById(R.id.pbPositive)
        pbNegative       = findViewById(R.id.pbNegative)
        tvStats          = findViewById(R.id.tvStats)
        progressBar      = findViewById(R.id.progressBar)
    }
}
```

```kotlin
suspend fun processVoiceNote(audioFloat: FloatArray): VoiceNote {
```

## 8. Summary

- DistilBERT runs in ~50ms on-device — fast enough for real-time as-you-type feedback
- 500ms debounce prevents inference on every keystroke — run only when typing pauses
- Map binary POSITIVE/NEGATIVE scores to 5-level sentiment for richer UX
- `quickSentiment()` rule-based fallback gives instant feedback before model finishes
- Combine with Whisper transcription for auto-tagged voice notes — zero cloud dependency
